"""
api/auth.py — OAuth 2.0 client_credentials.

Фичи:
  - requests.Session (keep-alive) — второй запрос в 2× быстрее
  - retry с экспоненциальной паузой на сетевые ошибки и 5xx
  - timeout=(connect, read) вместо единого
  - безопасное сообщение об ошибке (не печатает data целиком)
"""
from __future__ import annotations

import sys
import time

import requests

from api.errors import ApiAuthError
from config import CLIENT_ID, CLIENT_SECRET, OAUTH_URL

_session: requests.Session | None = None


def _get_session() -> requests.Session:
    """Ленивая инициализация Session — один на процесс."""
    global _session
    if _session is None:
        _session = requests.Session()
    return _session


def fetch_token(max_retries: int = 3, verbose: bool = False) -> dict:
    """
    POST /oauth/token (client_credentials).
    Возвращает dict с access_token и expires_in.

    Retry на:
      - ConnectionError, Timeout, 5xx
    Не retry на:
      - 401/403 (неверный client_secret — бессмысленно повторять)
      - 400 (неверный запрос)
    """
    last_err: Exception | None = None

    for attempt in range(max_retries):
        try:
            resp = _get_session().post(
                OAUTH_URL,
                data={
                    "client_id": CLIENT_ID,
                    "client_secret": CLIENT_SECRET,
                    "grant_type": "client_credentials",
                    "scope": "",
                },
                headers={"Content-Type": "application/x-www-form-urlencoded"},
                timeout=(10, 15),
            )
        except (requests.ConnectionError, requests.Timeout) as e:
            last_err = e
            if attempt < max_retries - 1:
                delay = 2 ** attempt
                if verbose:
                    sys.stderr.write(
                        f"  [auth] {type(e).__name__} — retry через {delay}s "
                        f"({attempt + 1}/{max_retries})\n"
                    )
                time.sleep(delay)
                continue
            break

        # 4xx — не retry, сразу ошибка
        if 400 <= resp.status_code < 500:
            if resp.status_code == 401:
                raise ApiAuthError(
                    "401 OAuth: неверный client_id или client_secret. "
                    "Проверьте .env / переменные окружения."
                )
            raise RuntimeError(
                f"OAuth {resp.status_code}: {resp.text[:200]}"
            )

        # 5xx — retry
        if resp.status_code >= 500:
            last_err = RuntimeError(f"OAuth {resp.status_code}: {resp.text[:200]}")
            if attempt < max_retries - 1:
                delay = 2 ** attempt
                if verbose:
                    sys.stderr.write(
                        f"  [auth] {resp.status_code} — retry через {delay}s "
                        f"({attempt + 1}/{max_retries})\n"
                    )
                time.sleep(delay)
                continue
            break

        # 200
        try:
            data = resp.json()
        except ValueError as e:
            last_err = e
            if attempt < max_retries - 1:
                time.sleep(2 ** attempt)
                continue
            break

        if "access_token" not in data:
            keys = list(data.keys()) if isinstance(data, dict) else type(data).__name__
            raise RuntimeError(f"Нет access_token в ответе (keys: {keys})")
        return data

    raise RuntimeError(
        f"Не удалось получить токен после {max_retries} попыток: {last_err}"
    )

=== api/characters.py ===

"""
api/characters.py — публичный профиль персонажа.
"""
from __future__ import annotations

import requests
from urllib.parse import quote

from api.errors import ApiAuthError
from config import EAPI_BASE


def fetch_character_profile(token: str, region: str, name: str) -> dict | None:
    """
    GET /{region}/character/by-name/{name}/profile

    Возвращает dict с профилем или None, если персонаж не найден (404).
    Бросает ApiAuthError при 401 (неверный токен).
    """
    safe_name = quote(name, safe="")
    url = f"{EAPI_BASE}/{region}/character/by-name/{safe_name}/profile"
    resp = requests.get(
        url,
        headers={
            "Authorization": f"Bearer {token}",
            "Accept": "application/json",
        },
        timeout=(10, 15),
    )
    if resp.status_code == 404:
        return None
    if resp.status_code == 401:
        raise ApiAuthError("401 — токен недействителен")
    resp.raise_for_status()
    return resp.json()

  Для каждого itemId цена берётся в два шага:

  1) /history (по умолчанию включён)
     GET /{region}/auction/{itemId}/history?limit=200
     → массив сделок с полями price, amount
     → unit_price = price / amount       (цена за штуку)
     → sorted, отсекаем 10%% с краёв      (trimmed median)
     → медиана середины                  (устойчиво к выбросам)
     Если сделок < 5 (min_samples) → шаг 2.

  2) /lots (fallback)
     GET /{region}/auction/{itemId}/lots?limit=200&additional=true
     → минимальная цена за штуку среди активных лотов
     Если лотов нет → цена = None.

  Флаг --no-history отключает шаг 1 (только /lots, в 2× быстрее).
  Флаг --no-fallback отключает шаг 2 (только /history).

  Пример:
    --no-history --no-fallback   → цена только из /history
    --no-history                 → цена только из /lots
    (по умолчанию)               → /history, если пуст → /lots
""")
    _p(HEADER)

def _topic_recursive() -> None:
    _p(HEADER)
    _p("  🔁 Рекурсивный крафт (--recursive)")
    _p(HEADER)
    _p("""
  Без --recursive:  каждый ингредиент покупается на рынке по его цене.
  С --recursive:    для каждого ингредиента выбирается min(рынок, крафт).

  Алгоритм (CraftCalculator.resolve):
    1) если item в исключениях → cost=0, source="excluded", выход
    2) market_cost = цена_рынка × amount
    3) craft_cost  = лучший рецепт из всех с result=item:
       - runs = ceil(amount / recipe.result.amount)
       - energy_cost = (recipe.energy × runs / 5000) × price(401j)
       - ingredients_cost = Σ resolve(ing, amount × runs)
       - total = ingredients_cost + energy_cost
    4) выбираем min(market_cost, craft_cost)
       при равенстве — рынок (проще, чем крафтить)

  Защита от циклов:
    path: frozenset[str] — если item уже в цепочке, крафт запрещён,
    fallback на рынок. Плюс --max-depth (12 по умолчанию).

  Мемоизация:
    (item_id, amount) → CostResult. Второй вызов той же пары мгновенный.

  В выводе:
    1r246  Хорог  ×4  [крафт]  × 3 370 ₽ = 13 479 ₽
      ↳ крафт: 2× (выход 4), энергия 1 760 ₽
        Фасолевый суп  ×2  [рынок]  × 2 000 ₽ = 4 000 ₽
""")
    _p(HEADER)

    _p(HEADER)
    _p("""
  ▸ «Прибыль: — (—)» и «Нет цен на N ингредиентов»
    Один или несколько ингредиентов не имеют ни истории, ни лотов.
    Шаги:
      1) main.py recipe <itemId> --recursive --explain-tree
         → покажет, какие базовые itemId без цены
      2) если это фармящееся сырьё (не торгуется) — используй --ignore-missing
         для нижней оценки прибыли или exclusions add <id> (если у вас есть)
      3) если это обычный предмет — попробуй --no-history (иногда /lots богаче)

  ▸ Долго висит / кажется, что зависло
    Heartbeat в stderr каждые 5 сек:
      💓 alive 12.3s | requests 47/500 | cache 47 | idle 0.3s | last: gq3k/history
    Если idle растёт — запрос висит на timeout (20 с). Ctrl+C безопасен.
    Флаги для ускорения:
      --no-history     (в 2× меньше запросов)
      --workers 8      (если API не против; следи за 429)
      --max-requests 200  (жёсткий лимит, чтобы не уйти в минуты)

  ▸ 429 Too Many Requests
    API Stalcraft не публикует rate-limit. Безопасно ~5 req/s.
    Снизить нагрузку:
      --workers 2 --max-requests 200 --no-history

  ▸ Ошибка 401 Unauthorized
    Токен протух (client_credentials живёт ~1 час).
    Просто перезапусти команду — новый токен будет получен автоматически.

  ▸ Ctrl+C во время prefetch
    Безопасно. Прогресс не сохраняется между запусками (кэш в памяти).

  ▸ Batch-анализ долгий
    С --recursive каждый рецепт может обходить 50–100 itemId.
    Уменьши количество рецептов:
      --max-recipes 10

def _clear() -> None:

def _ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        return ""


def _pause() -> None:
    _ask("\n  ↵ Enter для продолжения...")

class MenuSession:
    def __init__(self) -> None:
        self._item_index: dict | None = None
        self._hideout: dict | None = None
        self._price_service: PriceService | None = None
        self._region: str | None = None
        self._craft_calc: CraftCalculator | None = None
        self._excl_snapshot: frozenset[str] = frozenset()
        self._data_loaded = False
        self.opts = AnalysisOptions.menu_defaults()

    # ────────── ленивая инициализация ──────────
    def _ensure_data(self) -> None:
        if self._data_loaded:
            return
        with phase("Загрузка listing + рецептов"):
            listing = load_listing()
            self._item_index = build_item_index(listing)
            self._hideout = load_hideout_recipes()
        self._data_loaded = True

    def _ensure_price(self) -> None:
        self._ensure_data()
        if self._price_service is None or self._region != self.opts.region:
            with phase("OAuth: токен"):
                token = fetch_token()["access_token"]
            if self._price_service is not None:
                self._price_service.close()
            self._price_service = PriceService(
                token,
                region=self.opts.region,
                use_history=not self.opts.no_history,
                max_requests=self.opts.max_requests,
                heartbeat=False,
                verbose=False,
            )
            self._region = self.opts.region
            self._craft_calc = None
            self._excl_snapshot = frozenset()
    def _menu_character(self) -> None:
        _clear()
        name = _ask("Ник персонажа: ").strip()
        if not name:
            return
        region = _ask(f"Регион [{self.opts.region}]: ").strip().upper()
        if region:
            self.opts.region = region

        ctx = self._context()
        from core.runner import run_character
        try:
            run_character(name, self.opts, ctx)
        except Exception as e:
            print(f"\n[ERROR] {type(e).__name__}: {e}")
        _pause()

    def _menu_history(self) -> None:
        """Выбор категории → предмета → опции истории → вывод."""
        self._ensure_data()
        groups, cat_names, order = self._group_recipes()

        while True:  # категории
            _clear()
            print("── Категории (история покупок) ──\n")
            for i, cid in enumerate(order, 1):
                print(f"  {i:2d}. {cat_names[cid]}  ({len(groups[cid])})")
            print("   0. ← Назад")
            choice = _ask("\nВыбор: ")
            if choice in ("", "0"):
                return
            try:
                cid = order[int(choice) - 1]
            except (ValueError, IndexError):
                _pause()
                continue

            # уникальные предметы категории
            seen: dict[str, dict] = {}
            for r in groups[cid]:
                rid = r["result"][0]["item"]
                seen.setdefault(rid, r)
            items = sorted(seen.values(),
                           key=lambda r: get_item_name(
                               self._item_index, r["result"][0]["item"],
                               self.opts.lang))

            while True:  # предметы
                _clear()
                print(f"── {cat_names[cid]} — история покупок ──\n")
                for i, r in enumerate(items, 1):
                    name = get_item_name(self._item_index,
                                        r["result"][0]["item"], self.opts.lang)
                    print(f"  {i:3d}. {name}")
                print("\n    F. 🔍 Поиск по названию")
                print("    0. ← К категориям")
                choice = _ask("\nВыбор: ").strip()
                if choice in ("", "0"):
                    break
                if choice.upper() == "F":
                    q = _ask("Подстрока: ").lower()
                    if q:
                        filtered = [
                            r for r in items
                            if q in get_item_name(
                                self._item_index, r["result"][0]["item"],
                                self.opts.lang).lower()
                        ]
                        if filtered:
                            items = filtered
                        else:
                            print("  Ничего не найдено")
                            _pause()
                    continue
                try:
                    item_id = items[int(choice) - 1]["result"][0]["item"]
                except (ValueError, IndexError):
                    continue
                self._history_options(item_id)
                # после выхода возвращаемся к списку предметов

    def _history_options(self, item_id: str) -> None:
        limit = 200
        sort = "newest"
        days: int | None = None
        show = 30
        page = 0
        while True:
            _clear()
            name = get_item_name(self._item_index, item_id, self.opts.lang)
            print(f"── История: {name} ({item_id}) ──\n")
            print(f"  1. Лимит запроса: {limit}")
            print(f"  2. Сортировка: {sort}")
            print(f"  3. За последние дней: {days if days else 'все'}")
            print(f"  4. На странице: {show if show > 0 else 'все'}")
            print(f"  5. Страница: {page + 1}")
            print(f"  6. ✅ Показать историю")
            print(f"  0. ← К списку предметов")
            choice = _ask("\nДействие [0-6]: ")
            if choice == "1":
                v = _ask("Лимит (1-1000): ")
                try:
                    limit = max(1, min(1000, int(v)))
                except ValueError:
                    pass
            elif choice == "2":
                print("\n  1. newest  2. oldest  3. price-asc  4. price-desc")
                v = _ask("Выбор: ")
                mapping = {"1": "newest", "2": "oldest",
                           "3": "price-asc", "4": "price-desc"}
                if v in mapping:
                    sort = mapping[v]
            elif choice == "3":
                v = _ask("Дней (0 = все): ")
                try:
                    n = int(v)
                    days = n if n > 0 else None
                except ValueError:
                    pass
            elif choice == "4":
                v = _ask("Записей на странице (0 = все): ")
                try:
                    n = int(v)
                    show = n if n > 0 else 0
                    page = 0
                except ValueError:
                    pass
            elif choice == "5":
                v = _ask(f"Страница (1..N): ")
                try:
                    page = max(0, int(v) - 1)
                except ValueError:
                    pass
            elif choice == "6":
                self._run_history(item_id, limit, sort, days, show, page)
                _pause()
            elif choice in ("0", ""):
                return

    def _run_history(self, item_id: str, limit: int, sort: str,
                     days: int | None, show: int = 30, page: int = 0) -> None:
        _clear()
        ctx = self._context()
        from core.runner import run_history
        try:
            run_history(item_id, self.opts, ctx,
                        limit=limit, sort=sort, days=days,
                        show=show, page=page)
        except Exception as e:
            print(f"\n[ERROR] {type(e).__name__}: {e}")

    def _ensure_craft(self) -> set[str]:
        self._ensure_price()
        exclusions = load_exclusions()
        if self.opts.recursive:
            if (self._craft_calc is None
                    or self._excl_snapshot != frozenset(exclusions)):
                self._craft_calc = CraftCalculator(

                    max_depth=self.opts.max_depth,
                    exclusions=exclusions,

                self._excl_snapshot = frozenset(exclusions)

            self._craft_calc = None

    def _context(self) -> AnalysisContext:
        exclusions = self._ensure_craft()
        return AnalysisContext(
            item_index=self._item_index,
            hideout=self._hideout,
            price_service=self._price_service,
            craft_calc=self._craft_calc,
            exclusions=exclusions,
        )

    def _shutdown(self) -> None:
        if self._price_service is not None:
            self._price_service.close()

    # ────────── главное меню ──────────
    def run(self) -> None:
        try:
            while True:
                _clear()
                print("══════════════════════════════════════════════════════")
                print("  🧪 Stalcraft Crafting Calculator — меню")
                print("══════════════════════════════════════════════════════\n")
                print("  1. 🧮 Посчитать крафт предмета")
                print("  2. 🏆 Топ по категориям")
                print("  3. 🎁 Управление исключениями")
                print("  4. 📖 Справка")
                print("  5. 🧑 Профиль персонажа")
                print("  6. 🧾 История покупок предмета")   
                print("  7. 🚪 Выход\n")                  
                choice = _ask("Выбор [1-7]: ")
                if choice == "1":
                    self._menu_recipe()
                elif choice == "2":
                    self._menu_batch()
                elif choice == "3":
                    self._menu_exclusions()
                elif choice == "4":
                    self._menu_help()
                elif choice == "5":
                    self._menu_character()
                elif choice == "6":
                    self._menu_history()                   
                elif choice in ("7", "q", ""):
                    break
        except KeyboardInterrupt:
            print()
        finally:
            self._shutdown()

    # ────────── вспомогательное ──────────
    def _recipe_level(self, recipe: dict) -> int:
        perks = recipe.get("requirements", {}).get("perks", {})
        return max(perks.values()) if perks else 0

    def _group_recipes(self):
        from config import EXCLUDED_TOP_CATEGORIES
        groups: dict[str, list[dict]] = {}
        cat_names: dict[str, str] = {}
        order: list[str] = []
        for r in self._hideout["recipes"]:
            cid = extract_category_id(r) or "other"
            if cid in EXCLUDED_TOP_CATEGORIES:
                continue
            if cid not in groups:
                groups[cid] = []
                order.append(cid)
                cat_names[cid] = (r["category"]["lines"].get(self.opts.lang)
                                  or cid)
            groups[cid].append(r)
        return groups, cat_names, order

    # ────────── 🧮 recipe ──────────
    def _menu_recipe(self) -> None:
        self._ensure_data()
        groups, cat_names, order = self._group_recipes()

        while True:  # категории
            _clear()
            print("── Категории ──\n")
            for i, cid in enumerate(order, 1):
                print(f"  {i:2d}. {cat_names[cid]}  ({len(groups[cid])})")
            print("   0. ← Назад")
            choice = _ask("\nВыбор: ")
            if choice in ("", "0"):
                return
            try:
                cid = order[int(choice) - 1]
            except (ValueError, IndexError):
                _pause()
                continue

            sample_by_item: dict[str, dict] = {}
            levels_by_item: dict[str, int] = {}
            for r in groups[cid]:
                rid = r["result"][0]["item"]
                lvl = self._recipe_level(r)
                if rid not in sample_by_item or lvl < levels_by_item[rid]:
                    sample_by_item[rid] = r
                    levels_by_item[rid] = lvl

            all_item_ids = list(sample_by_item.keys())
            sort_mode = "alpha"
            filter_query = ""

            while True:  # предметы внутри категории
                def _sort_key(iid: str):
                    name = get_item_name(self._item_index, iid,
                                        self.opts.lang).lower()
                    if sort_mode == "level":
                        return (levels_by_item.get(iid, 0), name)
                    return (0, name)

                item_ids = sorted(all_item_ids, key=_sort_key)
                if filter_query:
                    item_ids = [
                        iid for iid in item_ids
                        if filter_query in get_item_name(
                            self._item_index, iid, self.opts.lang).lower()
                    ]

                _clear()
                sort_label = ("по алфавиту" if sort_mode == "alpha"
                              else "по уровню")
                filter_note = (f", фильтр: «{filter_query}»"
                               if filter_query else "")
                print(f"── {cat_names[cid]} ── "
                      f"({len(item_ids)} предметов, {sort_label}{filter_note})\n")

                if not item_ids:
                    print("  Ничего не найдено.")
                elif sort_mode == "level":
                    current_level = -1
                    for i, iid in enumerate(item_ids, 1):
                        lvl = levels_by_item.get(iid, 0)
                        if lvl != current_level:
                            current_level = lvl
                            if lvl:
                                print(f"\n  ── Уровень {lvl} ──")
                            else:
                                print("\n  ── Без перка ──")
                        r = sample_by_item[iid]
                        name = get_item_name(self._item_index, iid,
                                            self.opts.lang)
                        print(f"  {i:3d}. {name}  ×{r['result'][0]['amount']}")
                else:
                    for i, iid in enumerate(item_ids, 1):
                        r = sample_by_item[iid]
                        name = get_item_name(self._item_index, iid,
                                            self.opts.lang)
                        print(f"  {i:3d}. {name}  ×{r['result'][0]['amount']}")

                print()
                print("    A. 🔤 По алфавиту")
                print("    L. 📊 По уровню крафта")
                print("    F. 🔍 Поиск по названию")
                if filter_query:
                    print("    R. ↺ Сбросить фильтр")
                print("    0. ← К категориям")

"""
core/context.py — единый контекст анализа: данные + сервисы.
"""

from dataclasses import dataclass

from api.auth import fetch_token
from services.craft import CraftCalculator
from services.price import PriceService, phase
from storage.exclusions import load_exclusions
from storage.hideout import load_hideout_recipes
from storage.listing import build_item_index, load_listing


@dataclass
class AnalysisContext:
    item_index: dict
    hideout: dict
    price_service: PriceService
    craft_calc: CraftCalculator | None
    exclusions: set[str]

    def close(self) -> None:
        self.price_service.close()


def build_context(opts) -> AnalysisContext:
    """Собирает контекст: данные + токен + сервисы."""
    with phase("Загрузка listing + рецептов"):
        listing = load_listing()
        item_index = build_item_index(listing)
        hideout = load_hideout_recipes()

    with phase("OAuth: получение токена"):
        token = fetch_token()["access_token"]

    exclusions = load_exclusions()
    if opts.exclude:
        exclusions.update(
            x.strip() for x in opts.exclude.split(",") if x.strip()
        )

    price_service = PriceService(
        token,
        region=opts.region,
        fallback_to_lots=not opts.no_fallback,
        use_history=not opts.no_history,
        max_requests=opts.max_requests,
        heartbeat=not opts.no_heartbeat,
        verbose=opts.verbose,
    )
    craft_calc = None
    if opts.recursive:
        craft_calc = CraftCalculator(
            hideout, price_service,
            max_depth=opts.max_depth,
            exclusions=exclusions,
        )
    return AnalysisContext(
        item_index=item_index, hideout=hideout,
        price_service=price_service, craft_calc=craft_calc,
        exclusions=exclusions,
    )

def fmt_price(v: float | None) -> str:
    if v is None:
        return "—"
    return f"{round(v):,} ₽".replace(",", " ")


def fmt_pct(v: float | None) -> str:
    return f"{v:+.1f}%" if v is not None else "—"


def print_ingredient_line(ing: IngredientCost, indent: int = 2,
                          explain: bool = False,
                          show_ids: bool = False) -> None:
    pad = " " * indent
    tag = {"history": "рынок", "market": "рынок", "lots": "рынок",
           "craft": "крафт", "none": "нет",
           "excluded": "ноль"}.get(ing.source, ing.source)
    is_abnormal = (ing.source in ("none", "excluded")
                   or ing.name.startswith("[неизвестный"))
    id_col = f"{ing.item_id:6s} " if (show_ids or is_abnormal) else "       "
    line = (f"{pad}{id_col}{ing.name:30s} × {ing.amount:3d}  "
            f"[{tag:6s}]  × {fmt_price(ing.unit_price):>12s} = "
            f"{fmt_price(ing.total_price):>12s}")
    if explain and ing.craft_path and ing.craft_path.why:
        line += f"   ← {ing.craft_path.why}"
    print(line)


def print_ingredient_tree(ing, item_index, indent=2, max_depth_display=2,
                          current=0, explain=False, show_ids=False) -> None:
    print_ingredient_line(ing, indent, explain=explain, show_ids=show_ids)
    if ing.craft_path and ing.craft_path.children and current < max_depth_display:
        cp = ing.craft_path
        energy_note = ""
        if cp.energy_cost and cp.energy_cost > 0:
            energy_note = f", энергия {fmt_price(cp.energy_cost)}"
        print(f"{' ' * (indent + 2)}↳ крафт: {cp.craft_runs}× "
              f"(выход {cp.craft_yield}){energy_note}")
        for child in cp.children:
            if child.unit_price is None:
                c_unit = c_total = None
            else:
                c_unit = round(child.unit_price)
                c_total = c_unit * child.amount
            child_ing = IngredientCost(
                item_id=child.item_id,
                name=get_item_name(item_index, child.item_id),
                amount=child.amount,
                unit_price=c_unit, total_price=c_total,
                source=child.source,
                craft_path=child,
            )
            print_ingredient_tree(child_ing, item_index,
                                  indent + 4, max_depth_display,
                                  current + 1, explain=explain,
                                  show_ids=show_ids)

def _sort_label(sort: str) -> str:
    return _SORT_LABEL.get(sort, sort)

def _fmt_datetime(iso_str: str | None) -> str:
    if not iso_str:
        return "—"
    try:
        dt = datetime.fromisoformat(iso_str.replace("Z", "+00:00"))
    except ValueError:
        return iso_str
    if _TZ is not None:
        dt = dt.astimezone(_TZ)
    return dt.strftime("%d.%m.%Y %H:%M")


# ──────────────────────────────────────────────────────────────────
#  Prefetch (внутренний)
# ──────────────────────────────────────────────────────────────────
def _direct_items(recipes: list[dict], exclusions: set[str]) -> set[str]:
    items = set() if ENERGY_ITEM_ID in exclusions else {ENERGY_ITEM_ID}
    for r in recipes:
        for res in r.get("result", []):
            items.add(res["item"])
        for ing in r.get("ingredients", []):
            if ing["item"] in exclusions:
                continue
            items.add(ing["item"])
    return items


def _prefetch(ctx, recipes: list[dict], opts) -> None:
    if ctx.craft_calc is not None:
        items = (ctx.craft_calc.collect_tree_items(recipes)
                 if opts.recursive
                 else _direct_items(recipes, ctx.exclusions))
    else:
        items = _direct_items(recipes, ctx.exclusions)

    todo = [i for i in items if not ctx.price_service.is_cached(i)]
    if not todo:
        print(f"→ Все {len(items)} itemId в кэше")
        return
    label = f"Prefetch цен ({len(todo)} itemId, workers={opts.workers})"
    print()
    with phase(label, label_width=55):
        def _progress(done, total, current_id):
            name = get_item_name(ctx.item_index, current_id)
            sys.stdout.write(f"\r   [{done:3d}/{total:3d}] {name[:40]:40s}")
            sys.stdout.flush()
        ctx.price_service.prefetch(todo, workers=opts.workers, progress=_progress)
    print()

def run_batch(opts, ctx, recipes: list[dict], *,
              per_category: bool = False,
              top: int | None = None,
              max_recipes: int | None = None,
              sort: str = "margin") -> None:
    """
    Пакетный анализ ПРЕДВАРИТЕЛЬНО отфильтрованного списка рецептов.
    Категории из EXCLUDED_TOP_CATEGORIES исключаются ВСЕГДА
    (и в per_category, и в общем топе).
    """
    if not recipes:
        print("Нет рецептов для анализа.")
        return

    # ─── фильтр неинтересных категорий (работает в обоих режимах) ───
    recipes = [
        r for r in recipes
        if (extract_category_id(r) or "other") not in EXCLUDED_TOP_CATEGORIES
    ]
    if not recipes:
        print("После фильтра не осталось рецептов.")
        return

    groups, cat_names, order = _group_recipes(recipes, opts.lang)

    # ─── лимиты ───
    if per_category:
        limit = max_recipes or 0
        if limit > 0:
            for cid in order:
                groups[cid] = groups[cid][:limit]
    else:
        limit = max_recipes if max_recipes is not None else 30
        if limit > 0 and len(recipes) > limit:
            print(f"⚠️  Ограничиваю до {limit} рецептов (--max-recipes), "
                  f"по всем категориям поровну.")
            flat = _round_robin(groups, order, limit)
            groups, cat_names, order = _group_recipes(flat, opts.lang)

    all_recipes = [r for cid in order for r in groups[cid]]

    if per_category:
        print(f"Категорий: {len(order)}, рецептов: {len(all_recipes)}")
    else:
        print(f"Найдено рецептов: {len(all_recipes)}")
    if ctx.exclusions:
        print(f"Исключений: {len(ctx.exclusions)}")

    _prefetch(ctx, all_recipes, opts)

    print(f"\n▶ Анализ {len(all_recipes)} рецептов"
          f"{' [recursive]' if opts.recursive else ''}")
    t0 = time.time()
    analyses = []
    for i, r in enumerate(all_recipes, 1):
        name = get_item_name(ctx.item_index, r["result"][0]["item"], opts.lang)
        sys.stdout.write(f"\r   [{i:3d}/{len(all_recipes):3d}] "
                         f"{name[:36]:36s} ({time.time()-t0:5.1f}s)")
        sys.stdout.flush()
        analyses.append(analyze_recipe(
            r, ctx.item_index, ctx.price_service,
            fee_pct=opts.fee, craft_calc=ctx.craft_calc, lang=opts.lang,
            ignore_missing=opts.ignore_missing,
            exclusions=ctx.exclusions,
        ))
    print(f"\r   ✓ готово за {time.time()-t0:.1f}s" + " " * 40)

    top_n = top if top is not None else (5 if per_category else 10)
    sort_key = ((lambda x: x.margin_pct) if sort == "margin"
                else (lambda x: x.profit))
    sort_label = _sort_label(sort)

    if per_category:
        for cid in order:
            cat = [a for a in analyses
                   if (extract_category_id(a.recipe) or "other") == cid]
            full = [a for a in cat if a.margin_pct is not None]
            full.sort(key=sort_key, reverse=True)
            print(f"\n{'═' * 66}")
            print(f"  🏆 {cat_names[cid]} — топ-{top_n} по {sort_label}")
            print(f"{'─' * 66}")
            if not full:
                print("  нет рецептов с полными ценами")
                continue
            for i, a in enumerate(full[:top_n], 1):
                _print_top_line(i, a)
    else:
        full = [a for a in analyses if a.margin_pct is not None]
        partial = [a for a in analyses if a.margin_pct is None]
        print(f"\nС полными ценами: {len(full)}, без данных: {len(partial)}")
        full.sort(key=sort_key, reverse=True)
        print(f"\n🏆 ТОП-{top_n} по {sort_label}:")
        for a in full[:top_n]:
            print(f"  {fmt_pct(a.margin_pct):>8s}  {a.result_name:36s} "
                  f"прибыль {fmt_price(a.profit):>12s}, "
                  f"себест. {fmt_price(a.total_cost):>12s}")

    print_stats(ctx.price_service, ctx.craft_calc)

def run_history(item_id: str, opts, ctx,
                limit: int = 200,
                sort: str = "newest",
                days: int | None = None,
                show: int = 30,
                page: int = 0) -> None:
    """История покупок предмета с статистикой и постраничным списком сделок."""
    history = ctx.price_service.get_history(item_id, limit)
    name = get_item_name(ctx.item_index, item_id, opts.lang)

    print(f"\n=== История покупок: {name} ({item_id}) ===")

    if not history:
        print("  Сделок не найдено (или история недоступна).")
        return

    # --- нормализация ---
    deals: list[dict] = []
    for d in history:
        try:
            price = float(d.get("price", 0))
            amount = int(d.get("amount", 0))
        except (TypeError, ValueError):
            continue
        if amount <= 0 or price <= 0:
            continue
        deals.append({
            "price": price,
            "amount": amount,
            "unit": price / amount,
            "time": d.get("time") or d.get("timestamp"),
        })

    if not deals:
        print("  Нет валидных сделок.")
        return

    # --- фильтр по дням ---
    if days is not None and days > 0:
        cutoff = datetime.now(timezone.utc) - timedelta(days=days)
        filtered = []
        for d in deals:
            if not d["time"]:
                continue
            try:
                dt = datetime.fromisoformat(d["time"].replace("Z", "+00:00"))
            except ValueError:
                continue
            if dt >= cutoff:
                filtered.append(d)
        deals = filtered
        if not deals:
            print(f"  Нет сделок за последние {days} дн.")
            return

    # --- сортировка ---
    if sort == "newest":
        deals.sort(key=lambda d: d["time"] or "", reverse=True)
    elif sort == "oldest":
        deals.sort(key=lambda d: d["time"] or "")
    elif sort == "price-asc":
        deals.sort(key=lambda d: d["unit"])
    elif sort == "price-desc":
        deals.sort(key=lambda d: d["unit"], reverse=True)

    # --- статистика ---
    units = sorted(d["unit"] for d in deals)
    n = len(units)
    trim = int(n * 0.1)
    core = units[trim: n - trim] if trim else units
    median_all = statistics.median(units) if units else 0.0
    median_trim = statistics.median(core) if core else 0.0
    total_rub = sum(d["price"] for d in deals)
    total_units = sum(d["amount"] for d in deals)

    print(f"  Сделок: {n}  |  единиц: {total_units}  |  "
          f"сумма: {fmt_price(total_rub)}")
    print("\n  💰 Цена за штуку:")
    print(f"     min:              {fmt_price(units[0])}")
    print(f"     max:              {fmt_price(units[-1])}")
    print(f"     median:           {fmt_price(median_all)}")
    print(f"     median (trim10%): {fmt_price(median_trim)}")

    # --- постраничный вывод ---
    if show <= 0:
        show_n = n
        page = 0
        total_pages = 1
    else:
        show_n = show
        total_pages = max(1, (n + show_n - 1) // show_n)
        page = max(0, min(page, total_pages - 1))

    start = page * show_n
    end = min(start + show_n, n)

    labels = {
        "newest": "последние",
        "oldest": "первые",
        "price-asc": "самые дешёвые",
        "price-desc": "самые дорогие",
    }
    label = labels.get(sort, "первые")

    if total_pages > 1:
        print(f"\n  📋 {label} ({start + 1}–{end} из {n}, "
              f"стр. {page + 1}/{total_pages}):")
    else:
        print(f"\n  📋 {label} {show_n} сделок:")

    for i, d in enumerate(deals[start:end], start + 1):
        when = _fmt_datetime(d["time"]) if d["time"] else "—"
        print(f"    {i:3d}. {when:20s}  {fmt_price(d['unit']):>10s}/шт  "
              f"× {d['amount']:4d}  = {fmt_price(d['price']):>12s}")

    if total_pages > 1:
        nav = []
        if page > 0:
            nav.append(f"--page {page - 1} (назад)")
        if page < total_pages - 1:
            nav.append(f"--page {page + 1} (вперёд)")
        if nav:
            print(f"\n  💡 {' · '.join(nav)}")

    if opts.verbose:
        print(f"\n  Первый raw-элемент: {history[0]}")

# ──────────────────────────────────────────────────────────────────
#  Вспомогательное
# ──────────────────────────────────────────────────────────────────
def _group_recipes(recipes, lang):
    groups: dict[str, list[dict]] = {}
    cat_names: dict[str, str] = {}
    order: list[str] = []
    for r in recipes:
        cid = extract_category_id(r) or "other"
        if cid not in groups:
            groups[cid] = []
            order.append(cid)
            cat_names[cid] = r["category"]["lines"].get(lang) or cid
        groups[cid].append(r)
    return groups, cat_names, order


def _round_robin(groups: dict[str, list[dict]], order: list[str],
                 limit: int) -> list[dict]:
    """Берёт по одному рецепту из каждой категории по кругу, пока не наберёт limit."""
    queues = {cid: list(groups[cid]) for cid in order}
    picked: list[dict] = []
    while len(picked) < limit and any(queues.values()):
        for cid in order:
            if queues[cid] and len(picked) < limit:
                picked.append(queues[cid].pop(0))
    return picked


def _print_top_line(i: int, a) -> None:
    gift = (f" 🎁{sum(a.excluded_summary.values())}"
            if a.excluded_summary else "")
    print(f"  {i:2d}. {fmt_pct(a.margin_pct):>8s}  "
          f"{a.result_name:32s} "
          f"прибыль {fmt_price(a.profit):>10s}, "
          f"себест. {fmt_price(a.total_cost):>10s}{gift}")

"""
main.py — тонкий CLI dispatcher.
Вся логика анализа — в core/runner.py.
"""

import argparse
import sys

import requests

from cli.help import print_operations, print_topic, print_unknown_topic
from config import DEFAULT_REGION
from core.context import build_context
from core.options import AnalysisOptions
from core.runner import run_batch, run_character, run_history, run_recipe
from api.errors import ApiAuthError, ApiRateLimitError
from storage.exclusions import (
    add_exclusions, clear_exclusions, load_exclusions, remove_exclusions,
)
from storage.hideout import filter_recipes
from storage.listing import build_item_index, get_item_name, load_listing


# ────────── команды ──────────
def cmd_help(args) -> int:
    topic = getattr(args, "topic", None)
    if not topic:
        print_operations()
        return 0
    if not print_topic(topic):
        print_unknown_topic(topic)
        return 1
    return 0


def cmd_menu(args) -> int:
    from cli.menu import run_menu
    return run_menu()


def cmd_exclusions(args) -> int:
    action = getattr(args, "action", None)
    if action == "list":
        items = load_exclusions()
        if not items:
            print("Исключений нет.")
            return 0
        item_index = build_item_index(load_listing())
        print(f"Исключено предметов: {len(items)} "
              f"(database/exclusions.json)")
        for iid in sorted(items):
            print(f"  {iid:8s} {get_item_name(item_index, iid)}")
        return 0
    if action == "add":
        items = add_exclusions(set(args.items))
        print(f"Добавлено. Всего исключений: {len(items)}")
        return 0
    if action == "remove":
        items = remove_exclusions(set(args.items))
        print(f"Удалено. Всего исключений: {len(items)}")
        return 0
    if action == "clear":
        clear_exclusions()
        print("Список исключений очищен.")
        return 0
    return 1


def cmd_character(args) -> int:
    # character использует только region, lang, verbose
    opts = AnalysisOptions(
        region=args.region,
        lang=args.lang,
        verbose=getattr(args, "verbose", False),
    )
    ctx = build_context(opts)
    try:
        run_character(args.name, opts, ctx)
    finally:
        ctx.close()
    return 0


def cmd_recipe(args) -> int:
    opts = AnalysisOptions.from_args(args)
    ctx = build_context(opts)
    try:
        run_recipe(args.item_id, opts, ctx)
    finally:
        ctx.close()
    return 0


def cmd_history(args) -> int:
    opts = AnalysisOptions.from_args(args)
    ctx = build_context(opts)
    try:
        run_history(args.item_id, opts, ctx,
                    limit=args.limit, sort=args.sort, days=args.days,
                    show=args.show, page=args.page)
    finally:
        ctx.close()
    return 0


def cmd_batch(args) -> int:
    opts = AnalysisOptions.from_args(args)
    ctx = build_context(opts)
    try:
        recipes = filter_recipes(ctx.hideout, bench=args.bench,
                                 category_id=args.category, perk_id=args.perk)
        if not recipes:
            print("Под фильтр не попал ни один рецепт.")
            return 1
        run_batch(
            opts, ctx, recipes,
            per_category=args.per_category,
            top=args.top,
            max_recipes=args.max_recipes,
            sort=args.sort,
        )
    finally:
        ctx.close()
    return 0


# ────────── entrypoint ──────────
def add_network_args(p: argparse.ArgumentParser) -> None:
    """Только сетевые параметры — для character."""
    p.add_argument("--region", default=DEFAULT_REGION)
    p.add_argument("--lang", default="ru")
    p.add_argument("--verbose", action="store_true")


def add_analysis_args(p: argparse.ArgumentParser) -> None:
    """Полный набор — для recipe, history, batch."""
    add_network_args(p)
    p.add_argument("--fee", type=float, default=5.0,
                   help="Комиссия аукциона, %% (по умолчанию 5)")
    p.add_argument("--recursive", action="store_true")
    p.add_argument("--max-depth", type=int, default=12)
    p.add_argument("--no-fallback", action="store_true")
    p.add_argument("--no-history", action="store_true")
    p.add_argument("--max-requests", type=int, default=500)
    p.add_argument("--workers", type=int, default=4)
    p.add_argument("--no-heartbeat", action="store_true")
    p.add_argument("--explain", action="store_true")
    p.add_argument("--explain-tree", action="store_true")
    p.add_argument("--ignore-missing", action="store_true")
    p.add_argument("--exclude", default="")
    p.add_argument("--show-ids", action="store_true")


def main() -> int:
    if len(sys.argv) == 1:
        from cli.menu import run_menu
        return run_menu()

    p = argparse.ArgumentParser(description="Stalcraft crafting calculator")
    sub = p.add_subparsers(dest="command", required=True)

    p_menu = sub.add_parser("menu", help="Интерактивное меню")
    p_menu.set_defaults(func=cmd_menu)

    p_help = sub.add_parser("help", help="Справочник команд и тем")
    p_help.add_argument("topic", nargs="?", default=None,
                        help="examples | prices | energy | fee | recursive | "
                             "troubleshoot | exclusions | menu | all")
    p_help.set_defaults(func=cmd_help)

    p_excl = sub.add_parser("exclusions",
                            help="Управление исключениями (нулевая стоимость)")
    excl_sub = p_excl.add_subparsers(dest="action", required=True)
    for name, help_text in [
        ("list", "Показать исключённые предметы"),
        ("add", "Добавить предметы"),
        ("remove", "Удалить предметы"),
        ("clear", "Очистить все исключения"),
    ]:
        pp = excl_sub.add_parser(name, help=help_text)
        if name in ("add", "remove"):
            pp.add_argument("items", nargs="+", help="itemId")
        pp.set_defaults(func=cmd_exclusions)

    p1 = sub.add_parser("recipe", help="Анализ рецептов по itemId")
    p1.add_argument("item_id")
    add_analysis_args(p1)
    p1.set_defaults(func=cmd_recipe)

    ph = sub.add_parser("history", help="История покупок предмета")
    ph.add_argument("item_id")
    ph.add_argument("--limit", type=int, default=200,
                    help="Сколько сделок запросить у API (по умолчанию 200)")
    ph.add_argument("--show", type=int, default=30,
                    help="Записей на странице (по умолчанию 30, 0 = все)")
    ph.add_argument("--page", type=int, default=0,
                    help="Номер страницы (0 — первая)")
    ph.add_argument("--sort",
                    choices=["newest", "oldest", "price-asc", "price-desc"],
                    default="newest")
    ph.add_argument("--days", type=int, default=None)
    add_analysis_args(ph)
    ph.set_defaults(func=cmd_history)

    pc = sub.add_parser("character", help="Публичный профиль персонажа по имени")
    pc.add_argument("name", help="Ник персонажа")
    add_network_args(pc)
    pc.set_defaults(func=cmd_character)

    p2 = sub.add_parser("batch", help="Пакетный анализ по фильтру")
    p2.add_argument("--bench", choices=["workbench", "laboratory_table",
                                        "kitchen_table"])
    p2.add_argument("--category")
    p2.add_argument("--perk")
    p2.add_argument("--top", type=int, default=None)
    p2.add_argument("--max-recipes", type=int, default=None)
    p2.add_argument("--sort", choices=["margin", "profit"], default="margin")
    p2.add_argument("--per-category", action="store_true")
    add_analysis_args(p2)
    p2.set_defaults(func=cmd_batch)

    args = p.parse_args()
    try:
        return args.func(args)
    except ApiAuthError as e:
        print(f"\n[401] Ошибка авторизации: {e}\n"
              "Проверьте OAuth-настройки и актуальность учётных данных.")
        return 2
    except ApiRateLimitError as e:
        print(f"\n[429] {e}\n"
              "API временно ограничил частоту запросов. Повторите операцию позже.")
        return 3
    except requests.HTTPError as e:
        print(f"\n[HTTP ERROR] {e.response.status_code}: {e.response.text[:300]}")
        return 1
    except KeyboardInterrupt:
        print("\n[INTERRUPTED]")
        return 130
    except Exception as e:
        print(f"\n[ERROR] {type(e).__name__}: {e}")
        return 1


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""
Собирает текстовые файлы проекта в один project_dump.txt.
Запуск из корня: python make_dump.py

Для файлов из папки database/ записывается не весь JSON,
а только первые DATABASE_SAMPLE_ITEMS элементов каждого списка —
чтобы дамп оставался компактным.
"""
from pathlib import Path
import json
import sys

OUTPUT_NAME = "project_dump.txt"

# сколько элементов каждого списка из database/ показывать в дампе
DATABASE_SAMPLE_ITEMS = 3

EXTENSIONS = {
    ".py", ".md", ".txt", ".json", ".toml", ".cfg", ".ini",
    ".yml", ".yaml", ".sql", ".html", ".css", ".js", ".ts",
}

# файлы без расширения / dotfiles, которые всё равно нужны
INCLUDE_NAMES = {
    ".env", ".env.example", ".env.local", ".env.dev", ".env.prod",
    ".gitignore", ".dockerignore", ".editorconfig", ".flake8",
    "Dockerfile", "dockerfile",
    "Makefile", "makefile",
    "LICENSE", "LICENSE.txt", "LICENSE.md",
}

EXCLUDE_DIRS = {
    ".venv", "venv", "__pycache__",
    ".git", ".idea", ".vscode",
    ".pytest_cache", ".mypy_cache", ".ruff_cache",
    "node_modules", "dist", "build",
}

# ВАЖНО: запятая после элемента — это кортеж из одного элемента
EXCLUDE_PATH_PARTS: set[tuple[str, ...]] = {
    ("history",),
}

MAX_FILE_BYTES = 2 * 1024 * 1024


def should_skip_dir(rel_parts: tuple[str, ...]) -> bool:
    if any(part in EXCLUDE_DIRS for part in rel_parts):
        return True
    for seq in EXCLUDE_PATH_PARTS:
        n = len(seq)
        for i in range(len(rel_parts) - n + 1):
            if rel_parts[i:i + n] == seq:
                return True
    return False


def is_database_file(rel_parts: tuple[str, ...]) -> bool:
    """True для файлов внутри database/ (по полному пути rel.parts)."""
    return bool(rel_parts) and rel_parts[0] == "database"


def _truncate_json_lists(data, sample_size: int):
    """
    Обрезает все списки верхнего уровня (или списки-значения в dict)
    до sample_size элементов.

    Возвращает (truncated_data, notes), где notes — список пометок
    вида «<путь>: N из M» для каждого обрезанного списка.
    """
    notes: list[str] = []

    def trunc(lst: list, label: str) -> list:
        if len(lst) > sample_size:
            notes.append(f"{label}: {sample_size} из {len(lst)}")
            return lst[:sample_size]
        return lst

    if isinstance(data, list):
        return trunc(data, "root"), notes
    if isinstance(data, dict):
        result = {}
        for key, value in data.items():
            result[key] = trunc(value, key) if isinstance(value, list) else value
        return result, notes
    return data, notes


def format_database_content(text: str) -> str:
    """
    Для файлов из database/: если содержимое — валидный JSON,
    обрезает все списки до DATABASE_SAMPLE_ITEMS и возвращает
    pretty-printed JSON с пометкой о том, что было обрезано.
    Иначе возвращает исходный текст.
    """
    try:
        data = json.loads(text)
    except json.JSONDecodeError:
        return text  # не JSON — оставляем как есть

    truncated, notes = _truncate_json_lists(data, DATABASE_SAMPLE_ITEMS)
    if not notes:
        return text  # обрезать нечего (файл уже маленький)

    body = json.dumps(truncated, ensure_ascii=False, indent=2)
    note_line = "  [truncated: " + "; ".join(notes) + "]"
    return note_line + "\n" + body


def collect_files(root: Path) -> list[Path]:
    files: list[Path] = []
    try:
        it = root.rglob("*", recurse_symlinks=True)   # Py 3.13+
    except TypeError:
        it = root.rglob("*")                          # Py <= 3.12

    for path in it:
        if not path.is_file():
            continue
        rel = path.relative_to(root)
        rel_parts = rel.parts[:-1]
        if should_skip_dir(rel_parts):
            continue
        if path == root / OUTPUT_NAME:                # не включать сам себя
            continue
        name_ok = path.name in INCLUDE_NAMES
        ext_ok = path.suffix.lower() in EXTENSIONS
        if not (name_ok or ext_ok):
            continue
        try:
            if path.stat().st_size > MAX_FILE_BYTES:
                print(f"[skip: large] {rel}", file=sys.stderr)
                continue
        except OSError:
            continue
        files.append(path)

    files.sort(key=lambda p: p.relative_to(root).as_posix().lower())
    return files


def main() -> None:
    root = Path(__file__).resolve().parent
    out_path = root / OUTPUT_NAME

    files = collect_files(root)
    if not files:
        print("Не найдено подходящих файлов.")
        return

    with out_path.open("w", encoding="utf-8", newline="\n") as out:
        for i, path in enumerate(files, 1):
            rel = path.relative_to(root)
            rel_str = rel.as_posix()
            out.write(f"\n\n=== {rel_str} ===\n\n")
            try:
                content = path.read_text(encoding="utf-8", errors="replace")
            except Exception as e:
                content = f"[не удалось прочитать: {e}]"

            # для JSON из database/ — только несколько элементов каждого списка
            if is_database_file(rel.parts) and path.suffix.lower() == ".json":
                content = format_database_content(content)

            out.write(content)
            print(f"[{i}/{len(files)}] {rel_str}")

    size_mb = out_path.stat().st_size / (1024 * 1024)
    print(f"\nГотово. Файлов: {len(files)}, размер: {size_mb:.2f} МБ")


if __name__ == "__main__":
    main()

# 🧪 STALCRAFT Crafting Calculator

Консольный калькулятор экономики крафта в **STALCRAFT**. Помогает оценить себестоимость предметов, сравнить покупку на аукционе с самостоятельным крафтом, рассчитать прибыль и изучить историю торгов.

Калькулятор использует данные аукциона STALCRAFT API, рецепты убежища и локальные справочники проекта. Дополнительно можно запустить сборщик истории сделок и снимков активных лотов, чтобы анализировать ликвидность и динамику рынка.

## Возможности

- **Расчёт рецептов:** сравнение стоимости покупки и крафта с учётом вложенных рецептов.
- **Экономика крафта:** комиссия аукциона и стоимость энергии через предмет «Батарея холодного синтеза» (401j).
- **Пакетный анализ:** поиск наиболее выгодных рецептов по верстаку, категории или перку.
- **Цены аукциона:** медианная цена сделок из истории с резервным переходом к активным лотам.
- **Исключения:** ингредиенты из собственного запаса можно учитывать с нулевой стоимостью.
- **История предмета:** статистика и постраничный просмотр сделок, полученных из API.
- **Сбор рыночных данных:** фоновое сохранение новых сделок и периодических снимков лотов.
- **Анализ ликвидности:** оценка оборота, глубины стакана, спреда и изменения цен по времени.
- **Интерактивное меню и CLI:** работу можно выполнять как через меню, так и командами.
- **Устойчивые сетевые запросы:** повторные попытки при сетевых ошибках и ответах сервера 5xx/429, обработка Retry-After, отдельные сообщения для ошибок авторизации и ограничения API.

## Команды

### Интерактивное меню

~~~bash
python main.py
# или
python main.py menu
~~~

В меню доступны основные сценарии: расчёт крафта, топ рецептов, управление исключениями, справка, профиль персонажа и история покупок. При выборе предмета можно настроить регион, комиссию и параметры расчёта.

### Расчёт рецепта

~~~bash
python main.py recipe <itemId> [параметры]
~~~

Примеры:

~~~bash
# Сравнить рынок и крафт рекурсивно
python main.py recipe zz1o2 --recursive

# Показать объяснение расчёта и дерево ингредиентов
python main.py recipe zz1o2 --recursive --explain --explain-tree

# Рассчитать с комиссией 7 процентов
python main.py recipe zz1o2 --fee 7

# Не обращаться к истории сделок, использовать активные лоты
python main.py recipe zz1o2 --no-history

# Показать itemId ингредиентов
python main.py recipe zz1o2 --show-ids
~~~

Без --recursive рассчитывается цена предмета на рынке. С --recursive калькулятор дополнительно раскрывает рецепты ингредиентов и выбирает более дешёвый вариант: купить или изготовить.

### Пакетный анализ рецептов

~~~bash
python main.py batch [параметры]
~~~

Примеры:

~~~bash
# Проанализировать рецепты и вывести лучшие по марже
python main.py batch --top 20

# Ограничить анализ верстаком
python main.py batch --bench workbench --top 10

# Отфильтровать рецепты по категории или перку
python main.py batch --category ammunition
python main.py batch --perk <perkId>

# Отдельный топ внутри каждой категории
python main.py batch --per-category --top 10

# Сортировать по абсолютной прибыли
python main.py batch --sort profit --max-recipes 30

# Учитывать вложенный крафт
python main.py batch --recursive --top 20
~~~

Допустимые значения --bench: workbench, laboratory_table, kitchen_table. Параметры --category и --perk принимают идентификаторы из данных рецептов.

### История покупок предмета через API

~~~bash
python main.py history <itemId> [параметры]
~~~

Примеры:

~~~bash
python main.py history zz1o2
python main.py history zz1o2 --limit 200 --show 50
python main.py history zz1o2 --sort price-asc --days 7
python main.py history zz1o2 --sort oldest --page 2
~~~

Параметры:

| Параметр | Назначение |
|---|---|
| --limit N | Сколько сделок запросить у API (от 1 до 200; по умолчанию 200) |
| --show N | Сколько записей показывать на странице (по умолчанию 30; 0 — все) |
| --page N | Номер страницы, начиная с 0 |
| --sort newest | Сначала новые сделки (по умолчанию) |
| --sort oldest | Сначала старые сделки |
| --sort price-asc | Сначала сделки с меньшей ценой |
| --sort price-desc | Сначала сделки с большей ценой |
| --days N | Ограничить выборку последними N днями |

### Публичный профиль персонажа

~~~bash
python main.py character <name>
python main.py character <name> --region RU --verbose
~~~

Команда запрашивает публичный профиль персонажа по имени. Доступность отдельных полей зависит от ответа API. Инвентарь, склад и деньги персонажа через используемый тип OAuth-доступа не предоставляются.

### Управление исключениями

~~~bash
python main.py exclusions list
python main.py exclusions add <itemId> [ещё_itemId ...]
python main.py exclusions remove <itemId> [ещё_itemId ...]
python main.py exclusions clear
~~~

Исключённые предметы считаются имеющимися в запасе и получают нулевую стоимость. Список сохраняется в database/exclusions.json и действует между запусками.

Для разового исключения только в текущем расчёте используйте --exclude:

~~~bash
python main.py recipe zz1o2 --recursive --exclude 7rp7,404p
~~~

Чтобы найти идентификаторы ингредиентов, добавьте --show-ids.

> **Важно:** калькулятор не проверяет фактическое количество предметов на вашем складе. Если исключить ингредиент, его стоимость будет считаться нулевой во всём требуемом количестве.

## Параметры анализа

Эти параметры доступны для команд recipe, history и batch, если не указано иное.

| Параметр | По умолчанию | Описание |
|---|---:|---|
| --region | STALCRAFT_REGION или RU | Регион API |
| --lang | ru | Язык названий |
| --fee | 5.0 | Комиссия аукциона в процентах |
| --recursive | выключен | Сравнивать покупку и крафт с раскрытием рецептов |
| --max-depth | 12 | Максимальная глубина вложенных рецептов |
| --no-fallback | выключен | Не переходить к активным лотам, если истории недостаточно |
| --no-history | выключен | Не запрашивать историю сделок; использовать активные лоты |
| --max-requests | 500 | Лимит HTTP-запросов на экземпляр сервиса в рамках запуска |
| --workers | 4 | Число потоков предварительной загрузки цен |
| --no-heartbeat | выключен | Отключить периодический статус выполнения |
| --verbose | выключен | Подробный вывод сетевых запросов и диагностики |
| --explain | выключен | Показать причины выбора варианта расчёта |
| --explain-tree | выключен | Показать дерево базовых ингредиентов |
| --ignore-missing | выключен | Продолжить оценку при отсутствии цены на часть ингредиентов |
| --exclude | пусто | Список разовых исключений через запятую |
| --show-ids | выключен | Показывать itemId рядом с названиями |

### Как рассчитывается стоимость

1. Для предмета определяется рыночная цена по истории сделок. Используется медиана цены за единицу; крайние значения частично отсекаются.
2. Если истории недостаточно и включён резервный источник, калькулятор проверяет активные лоты и берёт минимальную цену за единицу.
3. При включённом --recursive рассчитывается стоимость доступных рецептов изготовления ингредиентов.
4. К стоимости ингредиентов добавляется стоимость энергии. Энергия оценивается через покупку батарей холодного синтеза (401j, 5000 единиц энергии на предмет).
5. Сравниваются полная стоимость крафта и покупки на рынке. При равенстве выбирается рынок.
6. Прибыль от продажи рассчитывается с учётом указанной комиссии аукциона.

Цены кэшируются в памяти во время работы процесса. Это ускоряет повторные обращения к одним и тем же предметам в рамках одного запуска; кэш не является общим между отдельными запущенными процессами.

## Сбор и анализ истории рынка

Сборщик scripts.collect_history отдельно от калькулятора сохраняет историю сделок и снимки активных лотов. Он не запускается автоматически при обычном расчёте рецепта.

### Разовый сбор

~~~bash
# Собрать данные для одного или нескольких предметов
python -m scripts.collect_history --items zz1o2 rwol5 404p

# Собрать данные для всех предметов, уже присутствующих в локальной истории
python -m scripts.collect_history --auto

# Собрать данные для всех itemId из database/listing_craft.json
python -m scripts.collect_history --all-listing

# Посмотреть список предметов без OAuth-запросов и записи в файлы
python -m scripts.collect_history --all-listing --dry-run
~~~

Можно комбинировать --items, --auto и --all-listing; повторяющиеся идентификаторы удаляются.

### Режим демона

~~~bash
# Обходить весь список craftable-предметов каждые 30 минут
python -m scripts.collect_history --all-listing --every 1800 --verbose

# Собирать все предметы из listing_craft каждые 2 часа,
# запрашивать до 200 записей и хранить данные за последние 30 дней
python -m scripts.collect_history --all-listing --every 7200 --limit 200 --keep-days 30
~~~

Параметры сборщика:

| Параметр | Описание |
|---|---|
| --items ID [ID ...] | Явный список предметов |
| --auto | Все предметы, для которых уже есть локальные файлы истории |
| --all-listing | Все предметы из database/listing_craft.json |
| --every SECONDS | Повторять сбор с указанным интервалом; включает режим демона |
| --limit N | Лимит сделок/лотов на запрос (от 1 до 200; по умолчанию 200) |
| --region REGION | Регион API |
| --verbose | Подробные сообщения |
| --keep-days N | Удалять записи старше N дней |
| --dry-run | Показать выбранные предметы без авторизации и записи |

Остановка демона — Ctrl+C.

### Анализ собранных данных

~~~bash
# Анализ за последние 7 дней
python -m scripts.analyze zz1o2

# Анализ за 30 дней с названием в заголовке
python -m scripts.analyze zz1o2 --days 30 --name "Фасолевый суп"

# Показать все предметы, для которых собираются данные
python -m scripts.analyze --all
~~~

Анализ использует локальные JSONL-файлы и выводит количество сделок, объём торгов, дневной оборот, медианные характеристики активных лотов, глубину стакана, спред и статистику цен по часам и дням недели.

Для устойчивых выводов требуется накопить данные. При малом количестве сделок или снимков анализатор выводит предупреждение. Часовые срезы используют часовой пояс из STALCRAFT_TIMEZONE.

## Запуск через Docker

В репозитории есть Dockerfile и docker-compose.yml. Образ собирается на базе Python 3.12, приложение работает от непривилегированного пользователя appuser.

Создайте .env по инструкции выше, затем соберите образ:

~~~bash
docker compose build
~~~

### Разовый запуск CLI

~~~bash
# Интерактивное меню
docker compose run --rm cli

# Анализ рецепта
docker compose run --rm cli recipe zz1o2 --recursive

# История сделок
docker compose run --rm cli history zz1o2 --days 7
~~~

### Постоянный сборщик

Сервис collector в docker-compose.yml настроен для постоянного сбора данных по всем предметам из database/listing_craft.json:

STALCRAFT API принимает параметр limit не более 200 для этих конечных точек. Значения выше 200 приводят к HTTP 400 Bad Request.


| Настройка | Значение |
|---|---:|
| Список предметов | Все записи из listing_craft.json |
| Интервал между проходами | 7200 секунд (2 часа) |
| Лимит API на запрос | До 200 записей |
| Срок хранения локальной истории | 30 дней |
| Перезапуск после сбоя или перезагрузки Docker | Включён (unless-stopped) |

Для каждого предмета сборщик запрашивает историю завершённых сделок (/history) и активные лоты (/lots). Старые записи удаляются при обслуживании очередного прохода, если они старше 30 дней. В каждом снимке активных лотов сохраняется сводка и до 50 самых дешёвых предложений, а не полный список всех лотов.

#### Первый запуск или обновление сборщика

На Orange Pi перейдите в каталог проекта, обновите рабочую ветку и пересоберите контейнер:

~~~bash
cd ~/CZCraftCalc
git switch feature/refactor
git pull --ff-only origin feature/refactor
sudo docker compose up -d --build collector
~~~

Команда запускает collector в фоне. Терминал можно закрыть — контейнер продолжит работу. Файл .env и каталог database остаются на хосте и не удаляются при пересборке образа.

#### Проверка работы

~~~bash
# Состояние контейнеров
sudo docker compose ps

# Логи сборщика в реальном времени
sudo docker compose logs -f --tail=100 collector
~~~

В логах ищите сообщения о начале прохода, количестве новых сделок, снимков и ошибках. Первый проход может занять заметное время, так как обрабатывается весь список предметов.

#### Управление сборщиком

~~~bash
# Остановить
sudo docker compose stop collector

# Запустить снова после остановки
sudo docker compose start collector

# Перезапустить
sudo docker compose restart collector
~~~

Поскольку для collector задано restart: unless-stopped, запущенный контейнер автоматически поднимется после перезапуска Docker или Orange Pi. Если контейнер был остановлен вручную командой stop, после загрузки его нужно запустить командой start.

Оба сервиса используют общий том ./database:/app/database, поэтому локальная история доступна после перезапуска контейнеров и командам CLI. Не запускайте несколько сборщиков одновременно для одной и той же базы данных.

## Устранение неполадок

### Ошибка OAuth / HTTP 401

Проверьте STALCRAFT_CLIENT_ID и STALCRAFT_CLIENT_SECRET в .env. Убедитесь, что секрет актуален. Для CLI ошибка авторизации возвращается с кодом завершения 2.

### Ограничение API / HTTP 429

Сервис повторяет запросы с паузами и учитывает заголовок Retry-After, если сервер его прислал. После исчерпания повторных попыток выводится ошибка ограничения API; CLI завершает работу с кодом 3.

При частых ограничениях уменьшите параллелизм (--workers 2), сократите объём анализа или увеличьте интервал фонового сбора. Не запускайте несколько сборщиков на одну базу данных.

### Нет цены на ингредиент

- Запустите расчёт с --explain-tree, чтобы найти ингредиенты без цены.
- Проверьте, торгуется ли предмет на аукционе.
- Попробуйте --no-history, чтобы использовать активные лоты вместо истории.
- Для предметов из собственного запаса используйте исключения.
- --ignore-missing позволяет получить оценку при неполных данных, но результат может быть занижен.

### Кажется, что программа зависла

При включённом heartbeat в stderr периодически выводятся число запросов, размер кэша и время с последней активности. Для подробной диагностики запустите команду с --verbose. Долгий запрос может ожидать ответа API; прервать выполнение можно через Ctrl+C.

---

Данные аукциона и доступность API контролируются STALCRAFT/EXBO. Результаты расчёта являются оценкой по полученным рыночным данным и не гарантируют фактическую цену сделки.

def _since_iso(days: int) -> str:
    return (datetime.now(timezone.utc) - timedelta(days=days)).isoformat()


def _parse_iso(s: str | None) -> datetime | None:
    if not s:
        return None
    try:
        return datetime.fromisoformat(s.replace("Z", "+00:00"))
    except ValueError:
        return None


# ──────────────────────────────────────────────────────────────────
#  Ликвидность
# ──────────────────────────────────────────────────────────────────
def _liquidity(deals, snapshots, days):
    n_days = max(days, 1)
    units = sum(d["amount"] for d in deals)
    rub = sum(d["price"] for d in deals)
    units_per_day = units / n_days
    rub_per_day = rub / n_days

    # активные лоты — медиана по всем снапшотам
    snap_amounts = [s["summary"]["total_amount"] for s in snapshots]
    snap_lots = [s["summary"]["n_lots"] for s in snapshots]
    snap_min = [s["summary"]["min_unit"] for s in snapshots]
    snap_depth = [s["summary"]["depth_10pct_amount"] for s in snapshots]

    active_amount = statistics.median(snap_amounts) if snap_amounts else 0
    active_lots = statistics.median(snap_lots) if snap_lots else 0
    min_unit = statistics.median(snap_min) if snap_min else 0
    depth_amount = statistics.median(snap_depth) if snap_depth else 0

    median_sold = statistics.median([d["unit"] for d in deals]) if deals else 0

    days_to_sell = (active_amount / units_per_day) if units_per_day > 0 else None
    turnover = (units_per_day / (active_amount + units_per_day)
                if (active_amount + units_per_day) > 0 else None)
    spread_pct = ((min_unit - median_sold) / median_sold * 100
                  if median_sold > 0 and min_unit > 0 else None)

    # классификация
    if units_per_day >= 50 and (days_to_sell is None or days_to_sell <= 2):
        level = "🟢 ВЫСОКАЯ"
    elif units_per_day >= 5 and (days_to_sell is None or days_to_sell <= 7):
        level = "🟡 СРЕДНЯЯ"
    else:
        level = "🔴 НИЗКАЯ"

    return {
        "deals": len(deals), "units": units, "rub": rub,
        "deals_per_day": len(deals) / n_days,
        "units_per_day": units_per_day,
        "rub_per_day": rub_per_day,
        "active_amount": active_amount, "active_lots": active_lots,
        "min_unit": min_unit, "median_sold": median_sold,
        "depth_amount": depth_amount,
        "days_to_sell": days_to_sell, "turnover": turnover,
        "spread_pct": spread_pct, "level": level,
    }

def _best_time(deals, days):
    if not deals:
        return

    by_hour: dict[int, list[float]] = defaultdict(list)
    by_wd: dict[int, list[float]] = defaultdict(list)
    by_day: dict[str, list[float]] = defaultdict(list)

    for d in deals:
        dt = _parse_iso(d.get("time"))
        if not dt:
            continue
        local = dt.astimezone(_TZ) if _TZ is not None else dt
        by_hour[local.hour].append(d["unit"])
        by_wd[local.weekday()].append(d["unit"])
        by_day[local.strftime("%Y-%m-%d")].append(d["unit"])

    all_units = [d["unit"] for d in deals]
    overall = statistics.median(all_units) if all_units else 0

    print(f"\n━━━ ЛУЧШЕЕ ВРЕМЯ (часы: {_TZ_LABEL}) ━━━")
    print(f"  Общая медиана: {fmt_price(overall)}/шт\n")

    # --- по часам ---
    if by_hour:
        print("  По часам (только часы с ≥3 сделками):")
        print(f"    {'час':>4} {'сделок':>7} {'медиана':>12} {'откл.':>8}")
        shown = 0
        for h in range(24):
            prices = by_hour.get(h, [])
            if len(prices) < 3:
                continue
            shown += 1
            med = statistics.median(prices)
            delta = (med - overall) / overall * 100 if overall else 0
            marker = " ← купить" if delta < -3 else (" ← продать" if delta > 3 else "")
            print(f"    {h:>4} {len(prices):>7} {fmt_price(med):>12} "
                  f"{delta:>+7.1f}%{marker}")
        if shown == 0:
            print("    (мало данных — нужно ≥3 сделок на час)")

    # --- по дням недели ---
    if by_wd:
        unique_days = len(by_day)
        print(f"\n  По дням недели (дней в выборке: {unique_days}):")
        names = ["Пн", "Вт", "Ср", "Чт", "Пт", "Сб", "Вс"]
        shown = 0
        for wd in range(7):
            prices = by_wd.get(wd, [])
            if len(prices) < 3:
                continue
            shown += 1
            med = statistics.median(prices)
            delta = (med - overall) / overall * 100 if overall else 0
            marker = " ← купить" if delta < -3 else (" ← продать" if delta > 3 else "")
            print(f"    {names[wd]:>2} {len(prices):>5} {fmt_price(med):>12} "
                  f"{delta:>+7.1f}%{marker}")
        if shown == 0:
            print("    (мало данных — нужна выборка за несколько дней недели)")

    # --- тренд по дням ---
    if len(by_day) >= 2:
        print(f"\n  Тренд по дням (последние {min(7, len(by_day))}):")
        days_sorted = sorted(by_day.keys())
        for day in days_sorted[-7:]:
            med = statistics.median(by_day[day])
            bar = "█" * max(1, int((med / overall) * 10)) if overall else ""
            print(f"    {day}  {fmt_price(med):>10}  {bar}")

"""
scripts/collect_history.py — сбор истории сделок и снапшотов лотов.

  python -m scripts.collect_history --items zz1o2 rwol5 404p --verbose
  python -m scripts.collect_history --auto --every 1800
"""
from __future__ import annotations

import argparse
import sys
import time
from datetime import datetime, timezone

from api.auth import fetch_token
from api.errors import ApiAuthError, ApiRateLimitError
from config import DEFAULT_REGION

    save_state, tracked_items, trim_old,
)
from services.price import PriceService


def _log(msg: str) -> None:
    ts = datetime.now(timezone.utc).strftime("%H:%M:%S")
    print(f"[{ts}] {msg}", flush=True)


def collect_once(price_service: PriceService, items: list[str],
                 limit: int = 200, verbose: bool = False) -> dict:
    state = load_state()
    summary = {"items": 0, "new_deals": 0, "snapshots": 0, "errors": 0}

    for item_id in items:
        summary["items"] += 1
        ts_now = datetime.now(timezone.utc).isoformat()

        # ─── 1. Deals ───
        try:
            history = price_service.get_history(item_id, limit)
        except (ApiAuthError, ApiRateLimitError):
            raise
        except Exception as e:
            summary["errors"] += 1
            _log(f"  ✗ {item_id}/history: {type(e).__name__}: {e}")
            history = []

        new_deals = []
        if history:
            seen = load_deal_hashes(item_id)
            for raw in history:
                norm = normalize_deal(item_id, raw)
                if norm is None or norm["_hash"] in seen:
                    continue
                seen.add(norm["_hash"])
                new_deals.append(norm)
            summary["new_deals"] += append_deals(item_id, new_deals)

        # ─── 2. Snapshot ───
        try:
            lots_data = price_service.get_lots(item_id, limit)
        except (ApiAuthError, ApiRateLimitError):
            raise
        except Exception as e:
            summary["errors"] += 1
            _log(f"  ✗ {item_id}/lots: {type(e).__name__}: {e}")
            lots_data = {"total": 0, "lots": []}

        snap = build_snapshot(item_id, lots_data["total"], lots_data["lots"])
        if snap:
            append_snapshot(item_id, snap)
            summary["snapshots"] += 1

        # ─── 3. State ───
        state[item_id] = {
            "last_run": ts_now,
            "last_deals": len(new_deals),
            "last_snapshot": bool(snap),
            "lots_total": lots_data["total"],
            "lots_sampled": len(lots_data["lots"]),
        }

        if verbose or new_deals or snap:
            extra = (f", snapshot lots={len(lots_data['lots'])}"
                     f" total={lots_data['total']}")
            _log(f"  · {item_id}: +{len(new_deals)} deals{extra}")

    save_state(state)
    return summary


def run_daemon(price_service, items, interval, limit, verbose,
               keep_days: int | None = None):
    _log(f"Демон: itemId={len(items)}, интервал={interval}с ({interval//60}мин)")
    try:
        while True:
            if keep_days is not None:
                td = ts = 0
                for iid in items:
                    d, s = trim_old(iid, keep_days)
                    td += d
                    ts += s
                if td or ts:
                    _log(f"trim(>{keep_days}д): -{td} deals, -{ts} snapshots")

            t0 = time.time()
            _log(f"▶ проход ({len(items)} itemId)")
            summary = collect_once(price_service, items, limit, verbose)
            _log(f"✓ items={summary['items']} "
                 f"new_deals={summary['new_deals']} "
                 f"snapshots={summary['snapshots']} "
                 f"errors={summary['errors']} ({time.time()-t0:.1f}s)")
            sleep_for = max(0, interval - (time.time() - t0))
            _log(f"⏸  пауза {sleep_for:.0f}с")
            time.sleep(sleep_for)
    except KeyboardInterrupt:
        _log("⏹  остановлено")


def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("--items", nargs="+", default=None,
                   help="itemId (можно несколько)")
    p.add_argument("--auto", action="store_true",
                   help="Все itemId, для которых уже есть история")
    p.add_argument("--all-listing", action="store_true",
                   help="Все itemId из database/listing_craft.json")
    p.add_argument("--every", type=int, default=None,
                   help="Режим демона: интервал в секундах")
    p.add_argument("--limit", type=int, default=200,
                   help="Сделок/лотов запрашивать у API (по умолчанию 200)")
    p.add_argument("--region", default=DEFAULT_REGION)
    p.add_argument("--verbose", action="store_true")
    p.add_argument("--keep-days", type=int, default=None,
                   help="Перед проходом удалять записи старше N дней")
    p.add_argument("--dry-run", action="store_true",
                   help="Только показать, что будет обработано")
    args = p.parse_args()

    # ─── сбор itemId ───
    items: list[str] = []
    if args.items:
        items.extend(args.items)
    if args.auto:
        items.extend(tracked_items())
    if args.all_listing:
        from config import LISTING_CRAFT_FILE
        from storage.listing import load_listing, extract_item_id
        listing = load_listing(LISTING_CRAFT_FILE)
        items.extend(
            iid for iid in (extract_item_id(x) for x in listing) if iid
            )
    items = list(dict.fromkeys(items))   # дедуп, с сохранением порядка

    if not items:
        print("Нет itemId: --items / --auto / --all-listing")
        return 1

    # ─── dry-run: без OAuth, без записи ───
    if args.dry_run:
        _log(f"dry-run: itemId={len(items)}")
        for iid in items[:20]:
            _log(f"  · {iid}")
        if len(items) > 20:
            _log(f"  … ещё {len(items) - 20}")
        return 0

    # ─── trim один раз (для режима демона он ещё раз сработает внутри) ───
    if args.keep_days is not None and not args.every:
        td = ts = 0
        for iid in items:
            d, s = trim_old(iid, args.keep_days)
            td += d
            ts += s
        if td or ts:
            _log(f"trim(>{args.keep_days}д): -{td} deals, -{ts} snapshots")

    # ─── OAuth и запуск ───
    try:
        token = fetch_token()["access_token"]
    except ApiAuthError as e:
        print(f"[FATAL] Ошибка авторизации OAuth: {e}")
        return 2
    except Exception as e:
        print(f"[FATAL] OAuth: {type(e).__name__}: {e}")
        return 1

    ps = PriceService(
        token, region=args.region,
        use_history=True, max_requests=10_000,
        heartbeat=False, verbose=args.verbose,
    )
    try:
        if args.every:
            run_daemon(ps, items, args.every, args.limit, args.verbose,
                       keep_days=args.keep_days)
            return 0
        summary = collect_once(ps, items, args.limit, args.verbose)
        _log(f"✓ items={summary['items']} new_deals={summary['new_deals']} "
             f"snapshots={summary['snapshots']} errors={summary['errors']}")
        return 0
    except ApiAuthError as e:
        _log(f"✗ Ошибка авторизации API: {e}")
        return 2
    except ApiRateLimitError as e:
        _log(f"✗ API ограничил частоту запросов: {e}")
        return 3
    finally:
        ps.close()


if __name__ == "__main__":
    sys.exit(main())

# ---------- результат resolve() для одного item/amount ----------
@dataclass
class CostResult:
    item_id: str
    amount: int
    cost: float | None
    source: str                     # "market" | "craft" | "excluded" | "none"
    unit_price: float | None

# ---------- сбор leaf-листьев выбранного пути (📦) ----------
def _collect_required(node: "CostResult",
                      out: dict[str, tuple[int, str]]) -> None:
    if node.source == "craft" and node.children:
        for child in node.children:
            _collect_required(child, out)
        return
    prev_amt, prev_src = out.get(node.item_id, (0, node.source))
    out[node.item_id] = (prev_amt + node.amount, prev_src)


# ---------- полное дерево: принудительно разворачиваем всё (🌳) ----------
_SRC_PRIORITY = {"history": 0, "market": 0, "lots": 0, "none": 1}


def _worse_source(a: str, b: str) -> str:
    return a if _SRC_PRIORITY.get(a, 99) >= _SRC_PRIORITY.get(b, 99) else b


def _collect_full_tree(item_id: str, amount: int,
                       calc: "CraftCalculator",
                       out: dict[str, tuple[int, str]],
                       path: frozenset[str],
                       depth: int) -> None:
    """
    Принудительно разворачивает рецепт до базовых ресурсов
    (игнорирует выбор resolve market vs craft).

    Исключённые предметы пропускаются — они уже есть на складе.
    """
    if item_id in calc.exclusions:
        return

    is_leaf = (
        item_id in path
        or depth > calc.max_depth
        or item_id not in calc.recipes_by_result
    )

    craft = None
    if not is_leaf:
        craft = calc._try_craft(item_id, amount, path, depth)
        if craft is None or not craft.children:
            is_leaf = True

    if is_leaf:
        info = calc.price_service.get(item_id)
        src = info.source if info.unit_price is not None else "none"
        prev, prev_src = out.get(item_id, (0, src))
        out[item_id] = (prev + amount, _worse_source(prev_src, src))
        return

    for child in craft.children:
        _collect_full_tree(child.item_id, child.amount, calc, out,
                           path | {item_id}, depth + 1)


class CraftCalculator:
    """Выбирает лучшее: купить на рынке или скрафтить. Учитывает энергию и исключения."""

    def __init__(self, hideout_data: dict, price_service: PriceService,
                 max_depth: int = 12, exclusions: set[str] | None = None):
        self.price_service = price_service
        self.max_depth = max_depth
        self.exclusions: set[str] = set(exclusions or [])
        self.recipes_by_result: dict[str, list[dict]] = {}
        for r in hideout_data["recipes"]:
            for res in r.get("result", []):
                self.recipes_by_result.setdefault(res["item"], []).append(r)
        self._memo: dict[tuple[str, int], CostResult] = {}
        self._resolve_calls = 0
        self._resolve_cache_hits = 0

    def resolve_stats(self) -> dict:
        return {
            "resolve_calls": self._resolve_calls,
            "cache_hits": self._resolve_cache_hits,
            "memo_size": len(self._memo),
            "exclusions": len(self.exclusions),
        }

    # ---------------- публичный API ----------------
    def resolve(self, item_id: str, amount: int,
                path: frozenset[str] = frozenset(),
                depth: int = 0) -> CostResult:
        self._resolve_calls += 1

        if item_id in self.exclusions:
            result = CostResult(
                item_id, amount, 0.0, "excluded", 0.0,
                ingredients_cost=0.0, energy_cost=0.0,
                why="исключён (нулевая стоимость)",
            )
            self._memo[(item_id, amount)] = result
            return result

        key = (item_id, amount)
        if item_id not in path and key in self._memo:
            self._resolve_cache_hits += 1
            return self._memo[key]

        price_info = self.price_service.get(item_id)
        market_unit = price_info.unit_price
        market_cost = market_unit * amount if market_unit is not None else None

        craft_result = None
        if (
            item_id not in path
            and depth < self.max_depth
            and item_id in self.recipes_by_result
        ):
            craft_result = self._try_craft(item_id, amount, path, depth)

        if market_cost is not None and craft_result is not None:
            if market_cost <= craft_result.cost:
                result = CostResult(
                    item_id, amount, market_cost, "market", market_unit,
                    ingredients_cost=market_cost, energy_cost=0.0,
                    market_cost=market_cost, craft_cost=craft_result.cost,
                    why=f"рынок {market_cost:.0f} ≤ крафт {craft_result.cost:.0f}",
                )
            else:
                craft_result.market_cost = market_cost
                craft_result.craft_cost = craft_result.cost
                craft_result.why = (f"крафт {craft_result.cost:.0f} < "
                                    f"рынок {market_cost:.0f}")
                result = craft_result
        elif market_cost is not None:
            why = ("только рынок (нет рецепта)"
                   if item_id not in self.recipes_by_result
                   else "только рынок (все рецепты недоступны)")
            result = CostResult(item_id, amount, market_cost, "market", market_unit,
                                ingredients_cost=market_cost, energy_cost=0.0,
                                market_cost=market_cost, why=why)
        elif craft_result is not None:
            craft_result.market_cost = None
            craft_result.craft_cost = craft_result.cost
            craft_result.why = "только крафт (нет цены на рынке)"
            result = craft_result
        else:
            why = ("нет цены и нет рецепта"
                   if item_id not in self.recipes_by_result
                   else "нет цены и все рецепты недоступны")
            result = CostResult(item_id, amount, None, "none", None, why=why)

        if item_id not in path:
            self._memo[key] = result
        return result

    # ---------------- энергия ----------------
    def _energy_cost(self, energy: float) -> float | None:
        if energy <= 0:
            return 0.0
        if ENERGY_ITEM_ID in self.exclusions:
            return 0.0
        info = self.price_service.get(ENERGY_ITEM_ID)
        if info.unit_price is None:
            return None
        return (energy / ENERGY_PER_ITEM) * info.unit_price

    # ---------------- рекурсивный крафт ----------------
    def _try_craft(self, item_id, amount, path, depth) -> CostResult | None:
        best: CostResult | None = None
        for recipe in self.recipes_by_result[item_id]:
            res = recipe["result"][0]
            yld = res["amount"]
            runs = math.ceil(amount / yld) if yld > 0 else 0
            if runs == 0:
                continue

            energy_total = recipe.get("energy", 0.0) * runs
            energy_cost = self._energy_cost(energy_total)
            if energy_cost is None:
                continue

            ingredients_sum = 0.0
            children: list[CostResult] = []
            ok = True
            for ing in recipe["ingredients"]:
                need = ing["amount"] * runs
                child = self.resolve(ing["item"], need, path | {item_id}, depth + 1)
                if child.cost is None:
                    ok = False
                    break
                ingredients_sum += child.cost
                children.append(child)
            if not ok:
                continue

            total = ingredients_sum + energy_cost
            if best is None or total < best.cost:
                best = CostResult(
                    item_id=item_id, amount=amount, cost=total, source="craft",
                    unit_price=total / amount,
                    craft_recipe=recipe, craft_runs=runs,
                    craft_yield=runs * yld, children=children,
                    ingredients_cost=ingredients_sum,
                    energy_cost=energy_cost,
                )
        return best

    # ---------------- explain-tree ----------------
    def explain_missing(self, item_id: str, depth: int = 0,
                        path: frozenset[str] = frozenset()) -> list[str]:
        if depth > 4 or item_id in path:
            return []
        if item_id in self.exclusions:
            return []
        recipes = self.recipes_by_result.get(item_id, [])
        if not recipes:
            return []
        problems: set[str] = set()
        for recipe in recipes:
            for ing in recipe["ingredients"]:
                ing_id = ing["item"]
                if ing_id in self.exclusions:
                    continue
                info = self.price_service.get(ing_id)
                if info.unit_price is not None:
                    continue
                if ing_id in self.recipes_by_result:
                    sub = self.explain_missing(ing_id, depth + 1, path | {item_id})
                    problems.update(sub if sub else {ing_id})
                else:
                    problems.add(ing_id)
        return sorted(problems)

    # ---------------- сбор дерева для prefetch ----------------
    def collect_tree_items(self, recipes: list[dict]) -> set[str]:
        seen_recipes: set[int] = set()
        items: set[str] = set()
        frontier = list(recipes)
        for _ in range(self.max_depth):
            nxt: list[dict] = []
            for r in frontier:
                rid = id(r)
                if rid in seen_recipes:
                    continue
                seen_recipes.add(rid)
                for res in r.get("result", []):
                    items.add(res["item"])
                for ing in r.get("ingredients", []):
                    if ing["item"] in self.exclusions:
                        continue
                    items.add(ing["item"])
                    for child in self.recipes_by_result.get(ing["item"], []):
                        nxt.append(child)
            if not nxt:
                break
            frontier = nxt
        if ENERGY_ITEM_ID not in self.exclusions:
            items.add(ENERGY_ITEM_ID)
        return items

def _display_prices(unit_price: float | None,
                    amount: int) -> tuple[float | None, float | None]:
    if unit_price is None:
        return None, None
    u = round(unit_price)
    return float(u), float(u * amount)


# ---------- анализ одного рецепта ----------
def analyze_recipe(
    recipe: dict,
    item_index: dict[str, dict],
    price_service: PriceService,
    fee_pct: float = 0.0,
    craft_calc: CraftCalculator | None = None,
    lang: str = "ru",
    ignore_missing: bool = False,
    exclusions: set[str] | frozenset[str] = frozenset(),
) -> RecipeAnalysis:
    result = recipe["result"][0]
    result_item_id = result["item"]
    result_amount = result["amount"]
    result_name = get_item_name(item_index, result_item_id, lang)

    result_info = price_service.get(result_item_id)
    revenue_gross = (result_info.unit_price * result_amount
                     if result_info.unit_price is not None else None)
    fee_amount = (revenue_gross * fee_pct / 100.0
                  if revenue_gross is not None else None)
    revenue_net = (revenue_gross - fee_amount
                   if revenue_gross is not None else None)

    ingredients: list[IngredientCost] = []
    ingredients_sum = 0.0
    missing = 0
    required_total: dict[str, tuple[int, str]] = {}   # 📦
    full_tree: dict[str, tuple[int, str]] = {}        # 🌳

    for ing in recipe["ingredients"]:
        name = get_item_name(item_index, ing["item"], lang)
        craft_path: CostResult | None = None

        if craft_calc is not None:
            cr = craft_calc.resolve(ing["item"], ing["amount"])
            _collect_required(cr, required_total)
            _collect_full_tree(ing["item"], ing["amount"], craft_calc,
                               full_tree, frozenset(), 0)
            if cr.cost is None:
                unit_disp = total_disp = None
                source = "none"
            else:
                source = cr.source
                unit_disp, total_disp = _display_prices(
                    cr.cost / ing["amount"], ing["amount"]
                )
            if cr.source == "craft" or cr.why:
                craft_path = cr
        elif ing["item"] in exclusions:
            unit_disp, total_disp = 0.0, 0.0
            source = "excluded"
            prev, prev_src = required_total.get(ing["item"], (0, "excluded"))
            required_total[ing["item"]] = (prev + ing["amount"], prev_src)
            # в full_tree исключённые не попадают
        else:
            info = price_service.get(ing["item"])
            if info.unit_price is None:
                unit_disp = total_disp = None
                source = "none"
                src_for_total = "none"
            else:
                # info.source = "history" | "lots" — для UI это всё "market"
                source = "market"
                src_for_total = "market"
                unit_disp, total_disp = _display_prices(
                    info.unit_price, ing["amount"]
                )
            prev, prev_src = required_total.get(ing["item"], (0, src_for_total))
            required_total[ing["item"]] = (prev + ing["amount"], prev_src)
            prev, prev_src = full_tree.get(ing["item"], (0, src_for_total))
            full_tree[ing["item"]] = (prev + ing["amount"], prev_src)

        if total_disp is None:
            missing += 1
        else:
            ingredients_sum += total_disp

        ingredients.append(IngredientCost(
            item_id=ing["item"], name=name, amount=ing["amount"],
            unit_price=unit_disp, total_price=total_disp,
            source=source, craft_path=craft_path,
        ))

    excluded_summary = {
        k: v[0] for k, v in required_total.items() if v[1] == "excluded"
    }

    if missing == 0:
        ingredients_cost = round(ingredients_sum)
        partial = False
    elif ignore_missing:
        ingredients_cost = round(ingredients_sum)
        partial = True
    else:
        ingredients_cost = None
        partial = False

    energy = recipe.get("energy", 0.0)
    energy_cost: float | None
    missing_energy = False
    if energy > 0:
        if ENERGY_ITEM_ID in exclusions:
            energy_cost = 0.0
        else:
            e_info = price_service.get(ENERGY_ITEM_ID)
            if e_info.unit_price is not None:
                energy_cost = round((energy / ENERGY_PER_ITEM) * e_info.unit_price)
            else:
                energy_cost = None
                missing_energy = True
    else:
        energy_cost = 0.0

    if ingredients_cost is not None and energy_cost is not None:
        total_cost = ingredients_cost + energy_cost
    else:
        total_cost = None

    cat = recipe["category"]["lines"].get(lang) or ""
    subcat = recipe["subcategory"]["lines"].get(lang) or ""

    if total_cost is None or revenue_net is None:
        return RecipeAnalysis(
            recipe=recipe, result_item_id=result_item_id,
            result_name=result_name, result_amount=result_amount,
            result_revenue_gross=revenue_gross, fee_pct=fee_pct,
            fee_amount=fee_amount, result_revenue_net=revenue_net,
            ingredients=ingredients,
            ingredients_cost=ingredients_cost,
            energy=energy, energy_cost=energy_cost,
            total_cost=None, profit=None, margin_pct=None,
            missing_ingredients=missing, missing_energy=missing_energy,
            partial=partial,
            excluded_summary=excluded_summary,
            required_total=required_total,
            full_tree=full_tree,
            category_ru=cat, subcategory_ru=subcat,
        )

    profit = revenue_net - total_cost
    margin = (profit / total_cost * 100) if total_cost > 0 else None

    return RecipeAnalysis(
        recipe=recipe, result_item_id=result_item_id,
        result_name=result_name, result_amount=result_amount,
        result_revenue_gross=revenue_gross, fee_pct=fee_pct,
        fee_amount=fee_amount, result_revenue_net=revenue_net,
        ingredients=ingredients,
        ingredients_cost=ingredients_cost,
        energy=energy, energy_cost=energy_cost,
        total_cost=total_cost, profit=profit, margin_pct=margin,
        missing_ingredients=missing, missing_energy=missing_energy,
        partial=partial,
        excluded_summary=excluded_summary,
        required_total=required_total,
        full_tree=full_tree,
        category_ru=cat, subcategory_ru=subcat,
    )

"""

def _deal_hash(item_id: str, raw: dict) -> str:
    raw_key = (f"{item_id}|{raw.get('time')}|"
               f"{raw.get('price')}|{raw.get('amount')}")
    return hashlib.sha1(raw_key.encode()).hexdigest()


def normalize_deal(item_id: str, raw: dict) -> dict | None:
    """Приводит raw-сделку из API к записи для файла. None если невалидна."""
    try:
        price = float(raw.get("price", 0))
        amount = int(raw.get("amount", 0))
    except (TypeError, ValueError):
        return None
    if price <= 0 or amount <= 0:
        return None
    return {
        "item": item_id,
        "price": price,
        "amount": amount,
        "unit": price / amount,
        "time": raw.get("time") or raw.get("timestamp"),
        "collected_at": datetime.now(timezone.utc).isoformat(),
        "_hash": _deal_hash(item_id, raw),
    }


def load_deal_hashes(item_id: str) -> set[str]:

            try:
                d = json.loads(line)
                h = d.get("_hash")
                if h:
                    hashes.add(h)
            except json.JSONDecodeError:
                continue
    return hashes

# ──────────────────────────────────────────────────────────────────

def normalize_lot(raw: dict) -> dict | None:
    """Компактная запись лота: amount, current, buyout, unit, start, end."""
    try:
        amount = int(raw.get("amount", 0))
        current = float(raw.get("currentPrice", 0) or 0)
        buyout = float(raw.get("buyoutPrice", 0) or 0)
    except (TypeError, ValueError):
        return None
    if amount <= 0:
        return None
    # цена для покупателя: buyout, если задан, иначе current
    effective = buyout if buyout > 0 else current
    if effective <= 0:
        return None
    return {
        "amount": amount,
        "current": current,
        "buyout": buyout,
        "unit": effective / amount,
        "start": raw.get("startTime"),
        "end": raw.get("endTime"),
    }


def build_snapshot(item_id: str, total: int, raw_lots: list[dict]) -> dict | None:
    """Собирает снапшот: summary + top-50 самых дешёвых лотов."""
    lots = [norm for raw in raw_lots if (norm := normalize_lot(raw)) is not None]
    if not lots:
        return None

    units = sorted(l["unit"] for l in lots)
    n = len(units)
    summary = {
        "n_lots": n,
        "total_amount": sum(l["amount"] for l in lots),
        "min_unit": units[0],
        "median_unit": units[n // 2],
        "max_unit": units[-1],
        "p25_unit": units[n // 4],
        "p75_unit": units[3 * n // 4],
    }

    # глубина: сколько лотов в пределах +10% от min
    depth_cutoff = units[0] * 1.1
    depth_lots = [l for l in lots if l["unit"] <= depth_cutoff]
    summary["depth_10pct_lots"] = len(depth_lots)
    summary["depth_10pct_amount"] = sum(l["amount"] for l in depth_lots)

    top = sorted(lots, key=lambda l: l["unit"])[:SNAPSHOT_LOTS_LIMIT]

    return {
        "ts": datetime.now(timezone.utc).isoformat(),
        "item": item_id,
        "total": int(total),                 # что отдал API в total
        "sampled": n,                        # сколько лотов пришло в ответе
        "summary": summary,
        "lots": top,
    }

def trim_old(item_id: str, days: int = 30) -> tuple[int, int]:
    """Удаляет записи старше N дней из deals и snapshots. Возвращает (d_removed, s_removed)."""
    from datetime import timedelta
    cutoff = (datetime.now(timezone.utc) - timedelta(days=days)).isoformat()

    d_removed = 0
    path = deals_path(item_id)
    if path.exists():
        kept = []
        for line in path.read_text(encoding="utf-8").splitlines():
            try:
                d = json.loads(line)
            except json.JSONDecodeError:
                continue
            if (d.get("time") or "") >= cutoff:
                kept.append(line)
            else:
                d_removed += 1
        path.write_text("\n".join(kept) + ("\n" if kept else ""), encoding="utf-8")

    s_removed = 0
    path = snapshots_path(item_id)
    if path.exists():
        kept = []
        for line in path.read_text(encoding="utf-8").splitlines():
            try:
                d = json.loads(line)
            except json.JSONDecodeError:
                continue
            if (d.get("ts") or "") >= cutoff:
                kept.append(line)
            else:
                s_removed += 1
        path.write_text("\n".join(kept) + ("\n" if kept else ""), encoding="utf-8")

    return d_removed, s_removed

=== services/price.py ===

"""
services/price.py — цены за штуку.

Фичи:
  - requests.Session (keep-alive)
  - потокобезопасный кэш
  - бюджет запросов (max_requests)
  - параллельный prefetch()
  - heartbeat в stderr
  - verbose: лог каждого HTTP-запроса
  - опционально: только /lots (без /history)
  - сырые данные для сборщика: get_history(), get_lots()
"""
from __future__ import annotations

import statistics
import sys
import threading
import time
from concurrent.futures import ThreadPoolExecutor, as_completed
from contextlib import contextmanager
from dataclasses import dataclass
from datetime import datetime, timezone
from email.utils import parsedate_to_datetime
from typing import Callable, Iterable

import requests

from api.errors import ApiAuthError, ApiRateLimitError
from config import DEFAULT_REGION, EAPI_BASE


@dataclass
class PriceInfo:
    item_id: str
    unit_price: float | None
    source: str          # "history" | "lots" | "none"
    samples: int = 0


class PriceService:
    def __init__(
        self,
        token: str,
        region: str = DEFAULT_REGION,
        ttl: int = 600,
        history_limit: int = 200,
        lots_limit: int = 200,
        trim_ratio: float = 0.1,
        min_samples: int = 5,
        fallback_to_lots: bool = True,
        use_history: bool = True,
        max_requests: int = 500,
        heartbeat: bool = True,
        heartbeat_interval: float = 5.0,
        verbose: bool = False,
        max_retries: int = 3,
        retry_backoff: float = 1.0,
        max_retry_delay: float = 30.0,
    ):
        self.token = token
        self.region = region
        self.ttl = ttl
        self.history_limit = history_limit
        self.lots_limit = lots_limit
        self.trim_ratio = trim_ratio
        self.min_samples = min_samples
        self.fallback_to_lots = fallback_to_lots
        self.use_history = use_history
        self.max_requests = max_requests
        self.heartbeat_enabled = heartbeat
        self.heartbeat_interval = heartbeat_interval
        self.verbose = verbose
        self.max_retries = max(1, int(max_retries))
        self.retry_backoff = max(0.0, float(retry_backoff))
        self.max_retry_delay = max(0.0, float(max_retry_delay))

        self._session = requests.Session()
        self._session.headers.update({
            "Authorization": f"Bearer {token}",
            "Accept": "application/json",
        })
        self._cache: dict[str, tuple[PriceInfo, float]] = {}
        self._lock = threading.Lock()
        self._closed = False
        self._requests_made = 0
        self._started_at = time.time()
        self._last_activity = time.time()
        self._last_activity_label = "init"

        self._hb_thread: threading.Thread | None = None
        self._hb_stop = threading.Event()
        if self.heartbeat_enabled:
            self._start_heartbeat()

    # ---------------- heartbeat ----------------
    def _start_heartbeat(self) -> None:
        self._hb_thread = threading.Thread(target=self._heartbeat_loop, daemon=True)
        self._hb_thread.start()

    def stop_heartbeat(self) -> None:
        self._hb_stop.set()
        if self._hb_thread is not None:
            self._hb_thread.join(timeout=1.0)

    def close(self) -> None:
        """Останавливает фоновые задачи и один раз закрывает HTTP-сессию."""
        with self._lock:
            if self._closed:
                return
            self._closed = True
        self.stop_heartbeat()
        self._session.close()

    def _heartbeat_loop(self) -> None:
        while not self._hb_stop.wait(self.heartbeat_interval):
            with self._lock:
                req = self._requests_made
                cache = len(self._cache)
                idle = time.time() - self._last_activity
                label = self._last_activity_label
            elapsed = time.time() - self._started_at
            sys.stderr.write(
                f"  💓 alive {elapsed:5.1f}s | requests {req}/{self.max_requests} | "
                f"cache {cache} | idle {idle:4.1f}s | last: {label}\n"
            )
            sys.stderr.flush()

    def _touch(self, label: str) -> None:
        with self._lock:
            self._last_activity = time.time()
            self._last_activity_label = label

    # ---------------- публичный API ----------------
    def get(self, item_id: str) -> PriceInfo:
        with self._lock:
            hit = self._cache.get(item_id)
            if hit and time.time() - hit[1] < self.ttl:
                return hit[0]

        info = self._fetch(item_id)
        with self._lock:
            self._cache[item_id] = (info, time.time())
        return info

    def get_lots(self, item_id: str, limit: int | None = None) -> dict:
        """
        Сырые активные лоты из /lots (для сборщика снапшотов).

        Возвращает {"total": int, "lots": list[dict]}.
        При ошибке — {"total": 0, "lots": []}.
        """
        limit = limit if limit is not None else self.lots_limit
        url = f"{EAPI_BASE}/{self.region}/auction/{item_id}/lots"
        r = self._http_get(
            url,
            {"limit": limit, "additional": "true"},
            f"{item_id}/lots",
        )
        if r is None or r.status_code != 200:
            return {"total": 0, "lots": []}
        try:
            data = r.json()
        except ValueError:
            return {"total": 0, "lots": []}
        try:
            total = int(data.get("total") or 0)
        except (TypeError, ValueError):
            total = 0
        return {
            "total": total,
            "lots": data.get("lots") or [],
        }

    def prefetch(
        self,
        item_ids: Iterable[str],
        workers: int = 4,
        progress: Callable[[int, int, str], None] | None = None,
    ) -> None:
        unique = list({i for i in item_ids if i})
        todo = [i for i in unique if not self._is_cached(i)]
        total = len(todo)
        if not total:
            return
        done = 0
        with ThreadPoolExecutor(max_workers=workers) as pool:
            futures = {pool.submit(self.get, iid): iid for iid in todo}
            for fut in as_completed(futures):
                iid = futures[fut]
                try:
                    fut.result()
                except (ApiAuthError, ApiRateLimitError):
                    raise
                except Exception as e:
                    if self.verbose:
                        sys.stderr.write(f"  ! fail {iid}: {e}\n")
                done += 1
                if progress:
                    progress(done, total, iid)

    def stats(self) -> dict:
        with self._lock:
            return {
                "requests": self._requests_made,
                "cache_size": len(self._cache),
                "budget_exhausted": self._requests_made >= self.max_requests,
                "max_requests": self.max_requests,
                "elapsed": time.time() - self._started_at,
            }

    def is_cached(self, item_id: str) -> bool:
        """Публичный алиас для _is_cached()."""
        return self._is_cached(item_id)

    # ---------------- внутреннее ----------------
    def _is_cached(self, item_id: str) -> bool:
        with self._lock:
            hit = self._cache.get(item_id)
            return bool(hit and time.time() - hit[1] < self.ttl)

    def _retry_delay(self, attempt: int, response=None) -> float:
        """Возвращает задержку по Retry-After или экспоненциальной схеме."""
        retry_after = None
        if response is not None:
            retry_after = response.headers.get("Retry-After")

        delay = None
        if retry_after is not None:
            try:
                delay = float(retry_after)
            except (TypeError, ValueError):
                try:
                    retry_at = parsedate_to_datetime(retry_after)
                    if retry_at.tzinfo is None:
                        retry_at = retry_at.replace(tzinfo=timezone.utc)
                    delay = (retry_at - datetime.now(timezone.utc)).total_seconds()
                except (TypeError, ValueError, OverflowError):
                    delay = None

        if delay is not None:
            # Не сокращаем паузу, которую явно запросил сервер.
            return max(0.0, delay)

        delay = self.retry_backoff * (2 ** attempt)
        return min(max(0.0, delay), self.max_retry_delay)

    def _http_get(self, url: str, params: dict, label: str) -> requests.Response | None:
        last_response = None
        for attempt in range(self.max_retries):
            with self._lock:
                if self._requests_made >= self.max_requests:
                    if self.verbose:
                        sys.stderr.write(f"  ! budget exhausted, skip {label}\n")
                    if last_response is not None and last_response.status_code == 429:
                        raise ApiRateLimitError(
                            f"429 — лимит запросов API исчерпан при обращении к {label}"
                        )
                    return None
                self._requests_made += 1
            self._touch(label)

            t0 = time.time()
            try:
                response = self._session.get(url, params=params, timeout=20)
            except requests.RequestException as e:
                if self.verbose:
                    dt = time.time() - t0
                    sys.stderr.write(f"  [GET] {label:22s} → ERR ({dt:.2f}s) "
                                     f"{type(e).__name__}\n")
                if attempt + 1 < self.max_retries:
                    delay = self._retry_delay(attempt)
                    if self.verbose:
                        sys.stderr.write(f"  [GET] повтор через {delay:.1f}s\n")
                    time.sleep(delay)
                    continue
                return None

            last_response = response
            if response.status_code == 401:
                raise ApiAuthError(
                    "401 — токен STALCRAFT API недействителен или истёк"
                )

            if response.status_code == 429:
                if attempt + 1 < self.max_retries:
                    delay = self._retry_delay(attempt, response)
                    if self.verbose:
                        sys.stderr.write(f"  [GET] {label:22s} → 429, повтор через {delay:.1f}s\n")
                    time.sleep(delay)
                    continue
                raise ApiRateLimitError(
                    f"429 — API ограничил частоту запросов для {label} "
                    f"после {self.max_retries} попыток"
                )

            if response.status_code >= 500 and attempt + 1 < self.max_retries:
                delay = self._retry_delay(attempt)
                if self.verbose:
                    sys.stderr.write(
                        f"  [GET] {label:22s} → {response.status_code}, "
                        f"повтор через {delay:.1f}s\n"
                    )
                time.sleep(delay)
                continue

            if self.verbose:
                dt = time.time() - t0
                sys.stderr.write(f"  [GET] {label:22s} → {response.status_code} ({dt:.2f}s)\n")
            return response

        return last_response
    def _fetch(self, item_id: str) -> PriceInfo:
        if self.use_history:
            median, samples = self._history_median(item_id)
            if median is not None:
                return PriceInfo(item_id, median, "history", samples)

        if self.fallback_to_lots:
            min_lot, lots_count = self._min_lot_price(item_id)
            if min_lot is not None:
                return PriceInfo(item_id, min_lot, "lots", lots_count)

        return PriceInfo(item_id, None, "none", 0)
    
    def get_history(self, item_id: str, limit: int | None = None) -> list[dict]:
        """
        Сырые сделки из /history.
        Возвращает [{price, amount, [time]}, ...] или [] при ошибке.
        """
        limit = limit if limit is not None else self.history_limit
        url = f"{EAPI_BASE}/{self.region}/auction/{item_id}/history"
        r = self._http_get(url, {"limit": limit}, f"{item_id}/history")
        if r is None or r.status_code != 200:
            return []
        try:
            data = r.json()
        except ValueError:
            return []
        return data.get("prices") or data.get("lots") or []

    def _history_median(self, item_id: str) -> tuple[float | None, int]:
        url = f"{EAPI_BASE}/{self.region}/auction/{item_id}/history"
        r = self._http_get(url, {"limit": self.history_limit}, f"{item_id}/history")
        if r is None or r.status_code != 200:
            return None, 0
        try:
            data = r.json()
        except ValueError:
            return None, 0

        prices = data.get("prices") or data.get("lots") or []
        unit = sorted(
            p["price"] / p["amount"]
            for p in prices
            if p.get("amount", 0) > 0 and p.get("price", 0) > 0
        )
        if len(unit) < self.min_samples:
            return None, len(unit)

        trim = int(len(unit) * self.trim_ratio)
        core = unit[trim: len(unit) - trim] if trim else unit
        return statistics.median(core), len(core)

    def _min_lot_price(self, item_id: str) -> tuple[float | None, int]:
        url = f"{EAPI_BASE}/{self.region}/auction/{item_id}/lots"
        r = self._http_get(url, {"limit": self.lots_limit, "additional": "true"},
                           f"{item_id}/lots")
        if r is None or r.status_code != 200:
            return None, 0
        try:
            data = r.json()
        except ValueError:
            return None, 0

        lots = data.get("lots") or data.get("prices") or []
        unit_prices = []
        for lot in lots:
            amount = lot.get("amount", 0)
            price = lot.get("buyoutPrice") or lot.get("currentPrice") or 0
            if amount > 0 and price > 0:
                unit_prices.append(price / amount)
        if not unit_prices:
            return None, 0
        return min(unit_prices), len(unit_prices)


@contextmanager
def phase(name: str, label_width: int = 40):
    """Печатает '▶ name ... ✓ X.Xs' в stderr."""
    t0 = time.time()
    dots = "." * max(3, label_width - len(name))
    sys.stderr.write(f"▶ {name} {dots} ")
    sys.stderr.flush()
    try:
        yield
    finally:
        dt = time.time() - t0
        sys.stderr.write(f"✓ {dt:.1f}s\n")
        sys.stderr.flush()

Формат: {"items": ["7rp7", "404p"]}

import json

from config import EXCLUSIONS_FILE


def load_exclusions() -> set[str]:
    if not EXCLUSIONS_FILE.exists():
        return set()
    try:
        with EXCLUSIONS_FILE.open(encoding="utf-8") as f:
            data = json.load(f)
    except (json.JSONDecodeError, OSError):
        return set()
    items = data.get("items", []) if isinstance(data, dict) else data
    return {str(i) for i in items if i}


def save_exclusions(items: set[str]) -> None:
    EXCLUSIONS_FILE.parent.mkdir(parents=True, exist_ok=True)
    with EXCLUSIONS_FILE.open("w", encoding="utf-8") as f:
        json.dump({"items": sorted(items)}, f, ensure_ascii=False, indent=2)


def add_exclusions(items: set[str]) -> set[str]:
    current = load_exclusions()
    current.update(items)
    save_exclusions(current)
    return current


def remove_exclusions(items: set[str]) -> set[str]:
    current = load_exclusions()
    current -= items
    save_exclusions(current)
    return current


def clear_exclusions() -> None:
    save_exclusions(set())

def extract_category_id(recipe: dict) -> str | None:
    """
    "hideout.benches.categories.ammunition.name" → "ammunition"
    """
    key = recipe.get("category", {}).get("key", "")
    parts = key.split(".")
    if "categories" in parts:
        idx = parts.index("categories")
        if idx + 1 < len(parts):
            return parts[idx + 1]
    return None


def find_recipes_by_result(data: dict, item_id: str) -> list[dict]:
    return [
        r for r in data["recipes"]
        if any(x.get("item") == item_id for x in r.get("result", []))
    ]


def filter_recipes(
    data: dict,
    bench: str | None = None,
    category_id: str | None = None,
    perk_id: str | None = None,
) -> list[dict]:
    out = []
    for r in data["recipes"]:
        if bench and r.get("bench") != bench:
            continue
        if category_id and extract_category_id(r) != category_id:
            continue
        if perk_id and perk_id not in r.get("requirements", {}).get("perks", {}):
            continue
        out.append(r)
    return out

"""

from __future__ import annotations

import json
from pathlib import Path

from config import LISTING_FILE


def load_listing(path: Path | None = None) -> list:
    """Читает JSON-массив листинга. По умолчанию — database/listing.json."""
    path = path or LISTING_FILE
    if not path.exists():
        raise FileNotFoundError(f"Не найден файл базы: {path}")
    with path.open(encoding="utf-8") as f:
        data = json.load(f)
    if not isinstance(data, list):
        raise ValueError(f"Ожидался JSON-массив в {path.name}")
    return data


def extract_item_id(item: dict) -> str | None:
    """
    Достаёт itemId из записи listing.
    Пример data: "/items/other/qyvk.json"  →  "qyvk"
    """
    raw = item.get("data")
    if not raw:
        return None
    return Path(raw).stem


def build_item_index(listing: list) -> dict[str, dict]:
    """itemId → запись listing."""
    index: dict[str, dict] = {}
    for item in listing:
        item_id = extract_item_id(item)
        if item_id:
            index[item_id] = item
    return index


def get_item_name(index: dict[str, dict], item_id: str, lang: str = "ru") -> str:
    """Возвращает название предмета по itemId. Если нет — сам itemId."""
    item = index.get(item_id)
    if not item:
        return f"[неизвестный itemId: {item_id}]"

    name_block = item.get("name", {})
    lines = name_block.get("lines", {})
    return (
        lines.get(lang)
        or lines.get("en")
        or lines.get("ru")
        or name_block.get("key")
        or item_id
    )

from storage.hideout import (
    extract_category_id,
    filter_recipes,
    find_recipes_by_result,
)


def _recipe(
    cat_key: str = "hideout.benches.categories.ammunition.name",
    bench: str = "workbench",
    perks: dict | None = None,
    result: str = "a",
    ingredients: list | None = None,
) -> dict:
    return {
        "bench": bench,
        "category": {"key": cat_key},
        "subcategory": {"key": "x"},
        "result": [{"item": result, "amount": 1}],
        "ingredients": ingredients or [],
        "requirements": {"perks": perks or {}},
    }


def test_extract_category_id():
    assert (
        extract_category_id(
            {"category": {"key": "hideout.benches.categories.ammunition.name"}}
        )
        == "ammunition"
    )
    assert extract_category_id({"category": {"key": "no.markers.here"}}) is None
    assert extract_category_id({}) is None


def test_find_recipes_by_result():
    data = {"recipes": [
        _recipe(result="a"),
        _recipe(result="b"),
        _recipe(result="a"),
    ]}
    assert len(find_recipes_by_result(data, "a")) == 2
    assert len(find_recipes_by_result(data, "z")) == 0


def test_filter_recipes_by_bench():
    data = {"recipes": [
        _recipe(bench="workbench"),
        _recipe(bench="laboratory_table"),
    ]}
    assert len(filter_recipes(data, bench="workbench")) == 1


def test_filter_recipes_by_category():
    data = {"recipes": [
        _recipe(cat_key="hideout.benches.categories.ammunition.name"),
        _recipe(cat_key="hideout.benches.categories.armor.name"),
    ]}
    assert len(filter_recipes(data, category_id="armor")) == 1


def test_filter_recipes_by_perk():
    data = {"recipes": [
        _recipe(perks={"ammunition": 1}),
        _recipe(perks={"armorer": 1}),
    ]}
    assert len(filter_recipes(data, perk_id="ammunition")) == 1

=== tests/test_storage_listing.py ===

from storage.listing import (
    build_item_index,
    extract_item_id,
    get_item_name,
)


def test_extract_item_id():
    assert extract_item_id({"data": "/items/other/qyvk.json"}) == "qyvk"
    assert extract_item_id({"data": "/items/misc/401j.json"}) == "401j"
    assert extract_item_id({}) is None
    assert extract_item_id({"data": None}) is None


def test_build_item_index():
    listing = [
        {"data": "/items/misc/401j.json"},
        {"data": "/items/misc/zzz.json"},
        {"data": None},  # пропускается
        {},              # пропускается
    ]
    idx = build_item_index(listing)
    assert set(idx) == {"401j", "zzz"}


def test_get_item_name_fallbacks():
    idx = {
        "a": {"name": {"lines": {"ru": "Ру", "en": "En"}}},
        "b": {"name": {"lines": {"en": "OnlyEn"}}},
        "c": {"name": {"key": "item.c.key", "lines": {}}},
    }
    assert get_item_name(idx, "a", "ru") == "Ру"
    assert get_item_name(idx, "a", "de") == "En"
    assert get_item_name(idx, "b", "ru") == "OnlyEn"
    assert get_item_name(idx, "c", "ru") == "item.c.key"
    assert get_item_name(idx, "nope", "ru") == "[неизвестный itemId: nope]"
